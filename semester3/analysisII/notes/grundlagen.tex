\section{Grundlagen}

\subsection{Grundlagen des $\mathbb{R}^n$}

\begin{definition}
    \textbf{Skalarprodukt:} Für $x, y \in \mathbb{R}^n$ ist das Standard-Skalarprodukt definiert als 
    $$\langle x, y \rangle = \sum_{i=1}^{n} x_i y_i$$
\end{definition}

\begin{definition}
    \textbf{Euklidische Norm \& Abstand:} Die Norm ist 
    $$|x| = \sqrt{\sum_{i=1}^{n} x_i^2}$$
    und der Abstand zweier Punkte 
    $$d(x,y) = |x-y|$$
\end{definition}

\begin{theorem}
    \textbf{Dreiecksungleichung:} Es gilt 
    $$|x-z| \le |x-y| + |y-z| \forall x, y, z \in \mathbb{R}^n$$
\end{theorem}

\subsection{Folgen in $\mathbb{R}^n$}

\begin{definition}
    Eine \textbf{Folge} ist eine Abbildung von $\mathbb{N}$ (oder $\mathbb{N}_0$) nach $\mathbb{R}^n$, wobei das Bild $x_n$ das $n$-te Glied ist.
\end{definition}

\begin{theorem}
    Eine Folge konvergiert gegen einen Grenzwert $y \in \mathbb{R}^n$ ($\lim_{n\to\infty} x_n = y$), wenn für jedes $\epsilon > 0$ ein $N > 0$ existiert, sodass $|x_n - y| < \epsilon$ für alle $n > N$ gilt. Andernfalls ist sie divergent.
\end{theorem}

\begin{lemma}
    Eine Folge konvergiert genau dann, wenn sie koordinatenweise konvergiert. Ihr Grenzwert ist eindeutig.
\end{lemma}

\subsection{Topologische Grundbegriffe}

\begin{definition}
    \textbf{Offener Ball:} Der offene Ball um $x$ mit Radius $r$ ist definiert als $B_r(x) = {y \in \mathbb{R}^n \mid |x-y| < r}$.
\end{definition}

\begin{definition}
    Eine Menge $A \subset \mathbb{R}^n$ ist \textbf{offen}, wenn jeder Punkt in $A$ einen vollständig in $A$ enthaltenen offenen Ball besitzt.
    Eine Menge ist \textbf{abgeschlossen}, wenn ihr Komplement ($X \setminus A$) offen ist.
\end{definition}

\begin{remark}
    Endliche Schnitte und beliebige Vereinigungen offener Mengen sind offen; endliche Vereinigungen und beliebige Schnitte abgeschlossener Mengen sind abgeschlossen.
\end{remark}

\begin{lemma}
    $A$ ist genau dann offen, wenn für jede konvergente Folge mit Grenzwert $x \in A$ ab einem gewissen Index $N$ alle Folgenglieder in $A$ liegen.
\end{lemma}

\begin{lemma}
    $A$ ist genau dann abgeschlossen, wenn der Grenzwert jeder konvergenten Folge, deren Glieder in $A$ liegen, ebenfalls in $A$ ist.
\end{lemma}

\begin{definition}
    \textbf{Inneres ($A^{\circ}$):} Die Vereinigung aller offenen Mengen, die in einer Teilmenge $A \subset \mathbb{R}^n$ enthalten sind (grösste offene Teilmenge).
\end{definition}

\begin{definition}
    \textbf{Abschluss ($\overline{A}$):} Der Schnitt aller abgeschlossenen Mengen, die die Teilmenge $A \subset \mathbb{R}^n$ enthalten (kleinste abgeschlossene Obermenge).
\end{definition}

\begin{definition}
    Der \textbf{(topologische) Rand} ist definiert als $\overline{A} \setminus A^\circ$.
\end{definition}

\subsection{Funktionen}

\begin{definition}
    Eine Funktion (Abbildung) ist eine Zuordnungsvorschrift, welche jedem zulässigen Input $\in \mathbb{R}^n$ genau einen Output aus dem Zielbereich (range) $\mathbb{R}^m$ zuordnet.
\end{definition}

\begin{definition}
    Es sei $U \subseteq \mathbb{R}^n$ offen, $f : U \to \mathbb{R}^m$ eine Funktion und $\tilde{x} \in U$. Falls für jede beliebige Folge $(x_k)_{k\in\mathbb{N}_0} \subseteq U$ mit $x_k \to \tilde{x}$ ein universelles $\tilde{y} \in \mathbb{R}^m$ existiert (unabhängig von der betrachteten Folge), sodass gilt
    $$f(x_k) \to \tilde{y}$$
    nennen wir $\tilde{y}$ den Grenzwert von $f$ für $x_k$ gegen $\tilde{x}$ und schreiben
    $$\lim_{x_k \to \tilde{x}} f(x_k) = \tilde{y}$$ 
\end{definition}

\begin{definition}
    Es sei $f : \mathbb{R}^n \to \mathbb{R}^m$ eine Funktion. $f$ heisst stetig, falls für alle $\tilde{x} \in \mathbb{R}^n$ gilt:
    $$\forall \epsilon > 0 \;\exists \delta > 0 \text{ s.t. } \|x - \tilde{x}\| \le \delta \implies \|f(x) - f(\tilde{x})\| \le \epsilon$$
\end{definition}

\begin{lemma}
    $f : \mathbb{R}^n \to \mathbb{R}^m$ setetig $\iff$ für jede Folge $(x_k)_{k\in\mathbb{N}_0} \subset \mathbb{R}^n$ mit $x_k \to x$ in $\mathbb{R}^n$ konvergiert die Folge $(f(x_k))_{k\in\mathbb{N}_0} \subset \mathbb{R}^m$ in $\mathbb{R}^m$ gegen $f(x)$.
\end{lemma}

\begin{definition}
    \textbf{Kompaktheit:} Eine Teilmenge $K \subset \mathbb{R}^n$ nennen wir kompakt, falls jede Folge $(x_s)_{s\in\mathbb{N}_0} \subset K$ in $K$ eine in $K$ konvergente Teilfolge besitzt.
\end{definition}

\subsection{Ableitungen}

\begin{definition}
    \textbf{Partielle Ableitung:} Sei $f: U \to \mathbb{R}$ eine Funktion auf einer offenen Menge $U \subset \mathbb{R}^n$ und $a \in U$. Die partielle Ableitung von $f$ nach der $i$-ten Koordinate im Punkt $a$ ist definiert als:
    $$ \frac{\partial f}{\partial x_i}(a) = \lim_{h \to 0} \frac{f(a_1, \dots, a_i + h, \dots, a_n) - f(a)}{h} $$
    falls dieser Grenzwert existiert. Sie beschreibt die Steigung der Funktion in Richtung der $i$-ten Basisrichtung.
\end{definition}

\begin{definition}
    \textbf{Richtungsableitung:} Allgemeiner lässt sich die Ableitung in eine beliebige Richtung $v \in \mathbb{R}^n$ (mit $\|v\|=1$) definieren als:
    $$ D_v f(a) = \lim_{h \to 0} \frac{f(a + h v) - f(a)}{h} $$
\end{definition}

\begin{remark}
    Falls $f$ in $a$ total differenzierbar ist, gilt $D_v f(a) = \nabla f(a) \cdot v$.
\end{remark}    

\begin{definition}
    \textbf{(totale) Ableitung:} Eine Funktion $f: U \to \mathbb{R}^m$ heisst (total) differenzierbar im Punkt $a \in U$, falls eine lineare Abbildung $L: \mathbb{R}^n \to \mathbb{R}^m$ existiert, sodass:
    $$ \lim_{x \to a} \frac{\|f(x) - f(a) - L(x - a)\|}{\|x - a\|} = 0 $$
    Die Abbildung $L$ heisst die (totale) Ableitung von $f$ in $a$, bezeichnet mit $df(a)$ oder $Df(a)$. Sie stellt die beste lineare Approximation von $f$ in der Nähe von $a$ dar.
\end{definition}

\begin{definition}
    \textbf{Jacobi-Matrix:} Falls $f: U \to \mathbb{R}^m$ in $a \in U$ differenzierbar ist, wird die lineare Abbildung $df(a)$ bezüglich der Standardbasen durch die $m \times n$ Jacobi-Matrix $J_f(a)$ (auch $Df(a)$) aller ersten partiellen Ableitungen dargestellt:
    $$ J_f(a) := \begin{pmatrix}
    \frac{\partial f_1}{\partial x_1}(a) & \frac{\partial f_1}{\partial x_2}(a) & \dots & \frac{\partial f_1}{\partial x_n}(a) \\
    \frac{\partial f_2}{\partial x_1}(a) & \frac{\partial f_2}{\partial x_2}(a) & \dots & \frac{\partial f_2}{\partial x_n}(a) \\
    \vdots & \vdots & \ddots & \vdots \\
    \frac{\partial f_m}{\partial x_1}(a) & \frac{\partial f_m}{\partial x_2}(a) & \dots & \frac{\partial f_m}{\partial x_n}(a)
    \end{pmatrix} = \begin{pmatrix} \nabla f_1(a)^T \\ \vdots \\ \nabla f_m(a)^T \end{pmatrix} = \left( \frac{\partial f_i}{\partial x_j}(a) \right)_{\substack{1 \le i \le m \\ 1 \le j \le n}} $$
    Die $i$-te Zeile entspricht dem transponierten Gradienten $\nabla f_i(a)^T$ der $i$-ten Komponentenfunktion $f_i$, und die $j$-te Spalte enthält die partiellen Ableitungen aller Komponenten nach $x_j$.
\end{definition}

\begin{definition}
    Für eine skalarwertige differenzierbare Funktion $f: U \to \mathbb{R}$ ($m=1$) ist der \textbf{Gradient} von $f$ in $a$ der Spaltenvektor aller partiellen Ableitungen:
    $$ \nabla f(a) = \left( \frac{\partial f}{\partial x_1}(a), \dots, \frac{\partial f}{\partial x_n}(a) \right)^T $$
\end{definition}

\begin{remark}
    Der Gradient zeigt in die Richtung des steilsten Anstiegs von $f$.
\end{remark}

\begin{theorem}
    \textbf{Stetige Differenzierbarkeit:} Eine Funktion $f: U \to \mathbb{R}^m$ ist genau dann (total) differenzierbar und die Ableitung stetig ($f \in C^1(U, \mathbb{R}^m)$), wenn alle partiellen Ableitungen $\frac{\partial f_i}{\partial x_j}$ auf $U$ existieren und dort stetig sind.
\end{theorem}

\begin{remark}
    \textbf{Höhere partielle Ableitungen:} Wenn die partiellen Ableitungen von $f$ selbst wieder differenzierbar sind, kann man zweite (und höhere) partielle Ableitungen bilden, notiert als:
    $$ \frac{\partial^2 f}{\partial x_i \partial x_j} = \frac{\partial}{\partial x_i} \left( \frac{\partial f}{\partial x_j} \right) $$
\end{remark}

\begin{definition}
    \textbf{$k$-fach stetig differenzierbar:} Eine Funktion heisst $k$-fach stetig differenzierbar (geschrieben $f \in C^k$), wenn alle partiellen Ableitungen bis zur $k$-ten Ordnung existieren und stetig sind. Ist $f$ beliebig oft differenzierbar (glatt), schreibt man $f \in C^\infty$.
\end{definition}

\begin{theorem}
    \textbf{Kettenregel:} Seien $f: U \subset \mathbb{R}^n \to \mathbb{R}^m$ in $a \in U$ und $g: V \subset \mathbb{R}^m \to \mathbb{R}^p$ (mit $f(U) \subset V$) in $y = f(a)$ total differenzierbar. Dann ist $g \circ f: U \to \mathbb{R}^p$ in $a$ differenzierbar, und es gilt für die Ableitungen:
    $$ D(g \circ f)(a) = Dg(f(a)) \circ Df(a) $$
    bzw. für die Jacobi-Matrizen:
    $$ J_{g \circ f}(a) = J_g(f(a)) \cdot J_f(a) = \begin{pmatrix}
    \frac{\partial g_1}{\partial y_1}(y) & \dots & \frac{\partial g_1}{\partial y_m}(y) \\
    \vdots & \ddots & \vdots \\
    \frac{\partial g_p}{\partial y_1}(y) & \dots & \frac{\partial g_p}{\partial y_m}(y)
    \end{pmatrix} \cdot \begin{pmatrix}
    \frac{\partial f_1}{\partial x_1}(a) & \dots & \frac{\partial f_1}{\partial x_n}(a) \\
    \vdots & \ddots & \vdots \\
    \frac{\partial f_m}{\partial x_1}(a) & \dots & \frac{\partial f_m}{\partial x_n}(a)
    \end{pmatrix} $$
\end{theorem}

\begin{theorem}
    \textbf{Satz von Schwarz:} Ist $f: U \to \mathbb{R}$ eine zweimal stetig differenzierbare Funktion ($f \in C^2(U)$), so ist die Reihenfolge der partiellen Ableitungen irrelevant. Für alle $i, j$ gilt:
    $$ \frac{\partial^2 f}{\partial x_i \partial x_j} = \frac{\partial^2 f}{\partial x_j \partial x_i} $$
\end{theorem}

\begin{remark}
    Das bedeutet, die Hesse-Matrix von $f$ ist symmetrisch.
\end{remark}

\begin{remark}
    \textbf{Multiindex und Fakultät:} Um höhere Ableitungen in mehreren Dimensionen kompakt zu schreiben, nutzt man Multiindizes $\alpha = (\alpha_1, \dots, \alpha_n) \in \mathbb{N}_0^n$.
    Man definiert:
    \begin{itemize}
        \item $|\alpha| = \alpha_1 + \dots + \alpha_n$ (Ordnung der Ableitung)
        \item $\alpha! = \alpha_1! \cdot \alpha_2! \dots \alpha_n!$ (Fakultät)
        \item $x^\alpha = x_1^{\alpha_1} x_2^{\alpha_2} \dots x_n^{\alpha_n}$
        \item $\partial^\alpha f = \frac{\partial^{|\alpha|} f}{\partial x_1^{\alpha_1} \dots \partial x_n^{\alpha_n}}$
    \end{itemize}
\end{remark}

\begin{definition}
    \textbf{Mehrdimensionale Taylorapproximation:} Sei $f: U \to \mathbb{R}$ eine $(k+1)$-mal stetig differenzierbare Funktion. Die Taylor-Approximation $k$-ten Grades von $f$ um den Entwicklungspunkt $a \in U$ lautet:
    $$ T_k(x) = \sum_{|\alpha| \le k} \frac{\partial^\alpha f(a)}{\alpha!} (x - a)^\alpha $$
\end{definition}

\begin{remark}
    \textbf{Intuition aus der Taylor-Approximation:} Aus der Taylor-Formel um den kritischen Punkt $x_0$ folgt wegen $\nabla f(x_0) = 0$:
    $$ f(x) - f(x_0) = \frac{1}{2} (x - x_0)^T H_f(x_0) (x - x_0) + o(\|x - x_0\|^2) $$
    Für kleine Abstände $v = x - x_0$ wird das Vorzeichen von $f(x) - f(x_0)$ ausschliesslich von der quadratischen Form $v^T H_f(x_0) v$ bestimmt:
    \begin{itemize}
        \item Positiv definit: $v^T H_f(x_0) v > 0$ für alle $v \neq 0 \implies f(x) > f(x_0)$ in allen Richtungen (nach oben offenes Paraboloid $\implies$ lokales Minimum).
        \item Negativ definit: $v^T H_f(x_0) v < 0$ für alle $v \neq 0 \implies f(x) < f(x_0)$ in allen Richtungen (nach unten offenes Paraboloid $\implies$ lokales Maximum).
        \item Indefinit: Krümmt sich in Richtungen mit positiven Eigenwerten nach oben und in Richtungen mit negativen Eigenwerten nach unten $\implies$ Bergsattel / Sattelpunkt.
    \end{itemize}
\end{remark}

\subsection{Hesse-Matrix, Laplace-Operator und Lokale Extrema}

\begin{definition}
    Sei $f: U \to \mathbb{R}$ eine zweimal partiell differenzierbare Funktion auf einer offenen Menge $U \subset \mathbb{R}^n$. Die \textbf{Hesse-Matrix} $H_f(x)$ (auch $D^2 f(x)$ oder $\nabla^2 f(x)$) ist die $n \times n$-Matrix aller zweiten partiellen Ableitungen:
    $$ H_f(x) := \begin{pmatrix} 
    \frac{\partial^2 f}{\partial x_1^2}(x) & \frac{\partial^2 f}{\partial x_1 \partial x_2}(x) & \dots & \frac{\partial^2 f}{\partial x_1 \partial x_n}(x) \\
    \frac{\partial^2 f}{\partial x_2 \partial x_1}(x) & \frac{\partial^2 f}{\partial x_2^2}(x) & \dots & \frac{\partial^2 f}{\partial x_2 \partial x_n}(x) \\
    \vdots & \vdots & \ddots & \vdots \\
    \frac{\partial^2 f}{\partial x_n \partial x_1}(x) & \frac{\partial^2 f}{\partial x_n \partial x_2}(x) & \dots & \frac{\partial^2 f}{\partial x_n^2}(x)
    \end{pmatrix} = \left( \frac{\partial^2 f}{\partial x_i \partial x_j}(x) \right)_{1 \le i, j \le n} $$
\end{definition}

\begin{theorem}
    Ist $f \in C^2(U)$ (zweimal stetig differenzierbar), so folgt aus dem Satz von Schwarz:
    $$ \frac{\partial^2 f}{\partial x_i \partial x_j}(x) = \frac{\partial^2 f}{\partial x_j \partial x_i}(x) \quad \forall i, j \in \{1, \dots, n\} \implies H_f(x) = H_f(x)^T$$
    Die Hesse-Matrix $H_f(x)$ ist somit eine \textbf{reelle symmetrische Matrix}.
\end{theorem}

\begin{remark}
    \textbf{Intuition zur Hesse-Matrix:} Die Hesse-Matrix ist das mehrdimensionale Analogon der zweiten Ableitung $f''(x)$. Während der Gradient $\nabla f(x)$ die Steigung und Orientierung der Tangentialebene bestimmt (lineare Näherung), erfasst die Hesse-Matrix die \textbf{Krümmung} der Fläche in alle Raumrichtungen (quadratische Näherung). Für jeden Richtungsvektor $v \in \mathbb{R}^n$ mit $\|v\|=1$ ist die zweite Richtungsableitung gegeben durch die quadratische Form:
    $$ D_v(D_v f)(x) = v^T H_f(x) v $$
\end{remark}

\begin{definition}
    Der \textbf{Laplace-Operator} $\Delta$ ordnet einer zweimal partiell differenzierbaren Funktion $f: U \to \mathbb{R}$ die Summe ihrer ungemischten zweiten partiellen Ableitungen zu:
    $$ \Delta f(x) := \sum_{i=1}^n \frac{\partial^2 f}{\partial x_i^2}(x) $$
    Er entspricht der Spur (Trace) der Hesse-Matrix sowie der Divergenz des Gradientenfeldes:
    $$ \Delta f(x) = \operatorname{tr}(H_f(x)) = \operatorname{div}(\nabla f(x)) = \nabla^2 f(x)$$
\end{definition}

\begin{remark}
    Eine Funktion $f \in C^2(U)$ heisst \textbf{harmonisch}, wenn sie die Laplace-Gleichung erfüllt:
    $$ \Delta f(x) = 0 \quad \forall x \in U $$
\end{remark}

\begin{remark}
    \textbf{Intuition zum Laplace-Operator:} $\Delta f(x)$ misst die lokale Krümmung gemittelt über alle Raumrichtungen bzw. die Abweichung des Funktionswertes $f(x)$ vom Mittelwert seiner Nachbarpunkte auf einer kleinen Kugeloberfläche um $x$:
    \begin{itemize}
        \item Ist $\Delta f(x) > 0$, so liegt $f(x)$ unter dem Durchschnitt seiner Nachbarn (talförmig / Senke).
        \item Ist $\Delta f(x) < 0$, so liegt $f(x)$ über dem Durchschnitt seiner Nachbarn (kuppelförmig / Quelle).
        \item Ist $\Delta f(x) = 0$ (harmonisch), so ist der Funktionswert in jedem Punkt exakt gleich dem Mittelwert seiner Umgebung (Mittelwerteigenschaft). Daher können harmonische Funktionen im Inneren eines Gebiets niemals isolierte lokale Maxima oder Minima besitzen (starkes Maximumprinzip).
    \end{itemize}
\end{remark}

\begin{definition}
    \textbf{Lokale und globale Extrema:} Sei $f: D \to \mathbb{R}$ eine Funktion auf $D \subseteq \mathbb{R}^n$ und $x_0 \in D$.
    \begin{itemize}
        \item $f$ hat in $x_0$ ein \textbf{lokales Minimum/Maximum}, wenn eine Umgebung $B_\epsilon(x_0)$ existiert, sodass:
        $$ f(x_0) \le \geq f(x) \quad \forall x \in B_\epsilon(x_0) \cap D $$
        \item Gilt die Ungleichung für alle $x \in (B_\epsilon(x_0) \cap D) \setminus \{x_0\}$ strikt so heisst das Extremum \textbf{strikt}.
        \item Gilt die Ungleichung für alle $x \in D$, so liegt ein \textbf{globales} Minimum bzw. Maximum vor.
    \end{itemize}
\end{definition}

\begin{definition}
    Sei $f: U \to \mathbb{R}$ auf einer offenen Menge $U \subset \mathbb{R}^n$ differenzierbar. Ein Punkt $x_0 \in U$ heisst \textbf{kritischer Punkt}, wenn der Gradient in $x_0$ verschwindet:
    $$ \nabla f(x_0) = 0 \iff \frac{\partial f}{\partial x_i}(x_0) = 0 \quad \forall i \in \{1, \dots, n\} $$
\end{definition}

\begin{remark}
    \textbf{Intuition:} In einem kritischen Punkt ist die Tangentialebene an den Funktionsgraphen vollkommen flach (horizontal): $z = f(x_0)$. Bei infinitesimalen Auslenkungen in jede beliebige Richtung $v$ verschwindet die Steigung: $D_v f(x_0) = \nabla f(x_0) \cdot v = 0$.
\end{remark}

\begin{definition}
    Eine reelle symmetrische $n \times n$-Matrix $A = A^T$ (und ihre quadratische Form $q(v) = v^T A v$) heisst:
    \begin{itemize}
        \item \textbf{positiv/negativ definit}, falls $v^T A v \gt \lt 0$ für alle $v \in \mathbb{R}^n \setminus \{0\}$.
        \item \textbf{positiv/negativ semidefinit}, falls $v^T A v \geq \le 0$ für alle $v \in \mathbb{R}^n$.
        \item \textbf{indefinit}, falls Vektoren $v, w \in \mathbb{R}^n$ existieren mit $v^T A v > 0$ und $w^T A w < 0$.
        \item \textbf{degeneriert}, falls $\det(A) = 0$.
    \end{itemize}
\end{definition}

\begin{remark}
    Eine symmetrische Matrix ist positiv (semi-)definit wenn für alle eigenwerte $\lambda$ gilt $\lambda (\geq)\gt 0$
\end{remark}

\begin{remark}
    \textbf{Definitheits-Schnelltest im $\mathbb{R}^2$ ($2 \times 2$-Hesse-Matrix):} Für $H = \begin{pmatrix} f_{xx} & f_{xy} \\ f_{xy} & f_{yy} \end{pmatrix}$ gilt mit $\det(H) = f_{xx} f_{yy} - f_{xy}^2$:
    \begin{itemize}
        \item $\det(H) > 0$ und $f_{xx} > 0 \implies H$ ist \textbf{positiv definit}.
        \item $\det(H) > 0$ und $f_{xx} < 0 \implies H$ ist \textbf{negativ definit}.
        \item $\det(H) < 0 \implies H$ ist \textbf{indefinit} (da $\det(H) = \lambda_1 \lambda_2 < 0$, haben die Eigenwerte unterschiedliche Vorzeichen).
        \item $\det(H) = 0 \implies H$ ist \textbf{degeneriert} (semidefinit; Eigenwert 0).
    \end{itemize}
\end{remark}

\begin{theorem}
    Sei $f \in C^2(U)$ und $x_0 \in U$ ein kritischer Punkt ($\nabla f(x_0) = 0$).
    \begin{enumerate}
        \item Ist die Hesse-Matrix $H_f(x_0)$ \textbf{positiv definit}, so besitzt $f$ in $x_0$ ein \textbf{striktes lokales Minimum}.
        \item Ist die Hesse-Matrix $H_f(x_0)$ \textbf{negativ definit}, so besitzt $f$ in $x_0$ ein \textbf{striktes lokales Maximum}.
        \item Ist die Hesse-Matrix $H_f(x_0)$ \textbf{indefinit und nicht degeneriert}, so besitzt $f$ in $x_0$ einen \textbf{Sattelpunkt} (kein lokales Extremum).
    \end{enumerate}
\end{theorem}

\begin{definition}
    Ein kritischer Punkt $x_0$ heisst \textbf{Sattelpunkt}, wenn er kein lokales Extremum ist, d.h. in jeder beliebig kleinen Umgebung $B_\epsilon(x_0)$ existieren Punkte $x_+, x_-$ mit:
    $$ f(x_-) < f(x_0) < f(x_+) $$
\end{definition}

\begin{remark}
    Wir können globale extrealstellen durch die bestimmung der kritischen punkte und das betrachten der funktion am rande des definitonsbereich finden.
\end{remark}
