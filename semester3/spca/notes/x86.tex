\section{Basic x86 Architecture and Assembly}

\subsection{Computer Architecture and the x86-64 Model}

\begin{definition} \textbf{Von Neumann Architecture:} 
The foundational computing model consisting of three primary components interconnected by system buses:
\begin{itemize}
    \item \textbf{Central Processing Unit (CPU):} Contains the \textbf{Control Unit} (fetches and decodes instructions), the \textbf{Arithmetic Logic Unit (ALU)} (executes arithmetic and logical operations), and the \textbf{Register File} (ultra-fast, low-latency internal storage).
    \item \textbf{Main Memory (RAM):} A linear sequence of byte-addressable storage holding both program instructions and runtime data.
    \item \textbf{Input/Output (I/O) System:} Interfaces the CPU and memory with external peripherals (disks, network, user interfaces).
\end{itemize}
\end{definition}

\begin{definition} \textbf{The Programmer-Visible Machine State (x86-64):} 
From the perspective of an assembly programmer or compiler, the state of an x86-64 processor comprises:
\begin{itemize}
    \item \textbf{Program Counter (\texttt{\%rip}):} The instruction pointer register storing the 64-bit virtual memory address of the next instruction to be fetched and executed.
    \item \textbf{General-Purpose Registers (GPRs):} 16 architectural 64-bit registers used to store integer data, function arguments, return values, and memory addresses.
    \item \textbf{Condition Code Flags (\texttt{RFLAGS}):} 1-bit status registers holding information about the arithmetic outcome of the most recently executed instruction (e.g., zero, negative, overflow, carry).
    \item \textbf{Vector / Floating-Point Registers:} Dedicated 128-bit to 512-bit registers (\texttt{\%xmm0}--\texttt{\%xmm15}, AVX \texttt{\%ymm}, \texttt{\%zmm}) for floating-point and SIMD operations.
    \item \textbf{Virtual Memory Space:} A byte-addressable array of memory locations accessed through 64-bit virtual addresses.
\end{itemize}
\end{definition}

\begin{remark} \textbf{From C to Executable Machine Code:} 
A high-level C program undergoes four distinct translation phases before execution:
\begin{align*}
\text{Source (\texttt{.c})} &\xrightarrow{\text{Preprocessor}} \text{Expanded (\texttt{.i})} \xrightarrow{\text{Compiler (\texttt{gcc -S})}} \text{Assembly (\texttt{.s})} \\
&\xrightarrow{\text{Assembler (\texttt{as})}} \text{Object (\texttt{.o})} \xrightarrow{\text{Linker (\texttt{ld})}} \text{Executable Binary}
\end{align*}
Assembly code is simply a human-readable textual representation of raw binary machine code instructions. While high-level languages abstract away hardware specifics (types, variables, expressions), assembly exposes the raw reality of hardware: fixed-size registers, memory addresses, condition codes, and explicit jump instructions.
\end{remark}

\subsection{Assembly Syntax: AT\&T vs. Intel}

\begin{remark} \textbf{AT\&T Syntax vs. Intel Syntax:} 
There are two major assembly syntax conventions for x86. The GNU toolchain on Linux (\texttt{gcc}, \texttt{as}, \texttt{gdb}, \texttt{objdump}) defaults to \textbf{AT\&T syntax}, whereas Windows and Intel documentation utilize \textbf{Intel syntax}.
\end{remark}

\begin{important}
In this course and throughout Linux systems programming, we strictly adhere to \textbf{AT\&T syntax}:
\[
\textbf{Mnemonic}\langle\text{size}\rangle \quad \textbf{Source}, \quad \textbf{Destination}
\]
For example, in \texttt{movq \%rax, \%rbx}, the value flows from left to right: from \texttt{\%rax} into \texttt{\%rbx}.
\end{important}

\subsection{Data Sizes and Register Organization}

\begin{definition} \textbf{x86-64 Data Sizes:} 
Originating from the 16-bit 8086 processor, Intel defines a \textbf{Word} as 16 bits (2 bytes). The size suffixes and corresponding C data types are:
\begin{itemize}
    \item \textbf{Byte (\texttt{b}):} 8 bits (1 byte) $\longleftrightarrow$ C \texttt{char}
    \item \textbf{Word (\texttt{w}):} 16 bits (2 bytes) $\longleftrightarrow$ C \texttt{short}
    \item \textbf{Double Word (\texttt{l} for long):} 32 bits (4 bytes) $\longleftrightarrow$ C \texttt{int}, 32-bit integers, \texttt{float}
    \item \textbf{Quad Word (\texttt{q}):} 64 bits (8 bytes) $\longleftrightarrow$ C \texttt{long}, 64-bit pointers (\texttt{void*}), \texttt{double}
\end{itemize}
\end{definition}

\begin{definition} \textbf{The 16 General-Purpose Registers:} 
An x86-64 CPU provides 16 general-purpose 64-bit registers. For backward compatibility with 32-bit (IA32), 16-bit (8086), and 8-bit architectures, lower sub-portions of each register can be accessed independently:
\end{definition}

\begin{important} \textbf{The 32-bit Zero-Extension Rule:} 
In x86-64, whenever an instruction generates a 32-bit register result (e.g., \texttt{movl}, \texttt{addl}, \texttt{xorl}), the CPU automatically \textbf{clears (sets to zero) the upper 32 bits} of the corresponding 64-bit register!
\begin{itemize}
    \item \texttt{movl \$0x12345678, \%eax} sets the entire \texttt{\%rax} to \texttt{0x0000000012345678}.
    \item Conversely, 8-bit and 16-bit operations (\texttt{movb}, \texttt{movw}) do \textbf{not} zero the upper bits; they preserve whatever was already present in the remaining higher bits of the register.
\end{itemize}
\end{important}

\subsection{Operand Specifiers and Memory Addressing Modes}

\begin{definition} \textbf{Operand Types:} Most instructions take one or two operands classified into three types:
\begin{enumerate}
    \item \textbf{Immediate (\$Imm):} A constant numerical literal encoded directly into the instruction bytes (e.g., \texttt{\$42}, \texttt{\$-16}, \texttt{\$0xFF}).
    \item \textbf{Register (\%reg):} The direct contents of one of the 16 registers (e.g., \texttt{\%rax}, \texttt{\%edx}).
    \item \textbf{Memory Access (\dots):} The value stored in memory at a computed virtual address, denoted by parentheses representing memory dereference.
\end{enumerate}
\end{definition}

\begin{definition} \textbf{General Addressing Form:} 
The most general addressing mode in x86-64 computes the effective memory address using the formula:
$$\text{Address} = \text{Imm} + [r_b] + [r_i] \times s$$
written syntactically in AT\&T format as:
$$\text{Imm}(r_b, r_i, s)$$
where:
\begin{itemize}
    \item $\text{Imm} \in \mathbb{R}$: An optional constant offset (displacement).
    \item $r_b$: The base register (any 64-bit general-purpose register).
    \item $r_i$: The index register (any 64-bit register \textbf{except} \texttt{\%rsp}).
    \item $s \in \{1, 2, 4, 8\}$: The scale factor (must be 1, 2, 4, or 8, exactly matching the byte size of primitive data types!).
\end{itemize}
\end{definition}

\begin{codeexample}
// Examples of addressing in assembly
movq (%rdi), %rax          // Dereference pointer in %rdi: rax = *rdi
movl 8(%rdi), %eax         // Read 4-byte int at offset 8: eax = *(rdi + 8)
movq (%rdi, %rcx, 8), %rax // Array indexing: rax = rdi[rcx] (array of 8-byte longs)
movl -16(%rbp), %eax       // Access local variable on the stack frame
\end{codeexample}

\subsection{Data Movement Instructions}

\begin{definition} \textbf{The \texttt{mov} Family:} Copies data from a source operand to a destination operand without altering the source:
$$\texttt{mov}\langle\text{size}\rangle \quad \text{Source}, \quad \text{Destination}$$
Available variants include \texttt{movb} (1 byte), \texttt{movw} (2 bytes), \texttt{movl} (4 bytes), and \texttt{movq} (8 bytes).
\end{definition}

\begin{important} \textbf{The Memory-to-Memory Prohibition:} 
An x86-64 instruction \textbf{cannot} perform both a memory read and a memory write simultaneously. Therefore, an instruction of the form:
$$\texttt{movq (\%rax), (\%rdx)} \quad \text{\textbf{IS ILLEGAL IN x86-64}}$$
To copy data from one memory location to another, the value must first be loaded into an intermediate register:
\begin{codeexample}
movq (%rax), %rcx    // Load from memory into temporary register %rcx
movq %rcx, (%rdx)    // Store from %rcx to destination memory
\end{codeexample}
\end{important}

\begin{definition} \textbf{Extension Instructions (Moving Smaller to Larger Destinations):} 
When moving data from a smaller source to a larger destination, the missing upper bits must be populated:
\begin{itemize}
    \item \textbf{Zero Extension (\texttt{movz}):} Fills remaining high-order bits with zeros (used for \textbf{unsigned} data types):
    \begin{itemize}
        \item \texttt{movzbl} (byte to doubleword), \texttt{movzbq} (byte to quadword), \texttt{movzwq} (word to quadword).
        \item Note: \texttt{movzlq} does not exist because standard \texttt{movl} already zeroes the upper 32 bits!
    \end{itemize}
    \item \textbf{Sign Extension (\texttt{movs}):} Replicates the most significant bit (the sign bit) into all upper bits (used for \textbf{signed} two's-complement data types):
    \begin{itemize}
        \item \texttt{movsbl}, \texttt{movsbq}, \texttt{movswq}, \texttt{movslq}.
        \item Special instruction: \texttt{cltq} sign-extends \texttt{\%eax} into the entire 64-bit \texttt{\%rax}.
    \end{itemize}
\end{itemize}
\end{definition}

\begin{definition} \textbf{Stack Operations (\texttt{pushq} and \texttt{popq}):} 
The runtime stack is managed by the stack pointer register \texttt{\%rsp}, which points to the current top of the stack. Because the x86 stack \textbf{grows downward} towards lower numerical memory addresses:
\begin{itemize}
    \item \textbf{\texttt{pushq Src}:} Pushes an 8-byte value onto the stack:
    $$\texttt{\%rsp} \leftarrow \texttt{\%rsp} - 8, \quad \text{Mem}[\texttt{\%rsp}] \leftarrow \text{Src}$$
    \item \textbf{\texttt{popq Dst}:} Pops the top 8-byte value from the stack into a destination register:
    $$\text{Dst} \leftarrow \text{Mem}[\texttt{\%rsp}], \quad \texttt{\%rsp} \leftarrow \texttt{\%rsp} + 8$$
\end{itemize}
\end{definition}

\subsection{Arithmetic and Logical Operations}

\subsubsection{Load Effective Address (\texttt{leaq})}

\begin{definition} \textbf{Load Effective Address (\texttt{leaq}):} 
The \texttt{leaq} instruction shares the exact memory-addressing syntax as \texttt{movq}, but with one crucial difference: \textbf{it does not access memory!}
$$\texttt{leaq} \quad \text{Imm}(r_b, r_i, s), \quad \text{Dst}$$
Instead of reading from memory, \texttt{leaq} computes the address calculation $\text{Imm} + [r_b] + [r_i] \times s$ and stores the resulting numerical value directly into \texttt{Dst}.
\end{definition}

\begin{remark} \textbf{The Dual Nature and Compiler Optimization of \texttt{leaq}:} 
Compilers use \texttt{leaq} for two distinct purposes:
\begin{enumerate}
    \item \textbf{Generating Pointers:} To compute memory addresses (equivalent to the C address-of operator \texttt{\&}), such as obtaining the address of a struct field: \texttt{leaq 8(\%rdi), \%rax}.
    \item \textbf{Fast Integer Arithmetic:} Because modern CPUs have dedicated hardware address generation units (AGUs) that compute $\text{base} + \text{index} \times \text{scale}$ in a single clock cycle, compilers hijack \texttt{leaq} to perform integer math without memory involvement.
\end{enumerate}
\end{remark}

\begin{codeexample}
// Fast arithmetic tricks performed by the compiler via leaq:
leaq (%rdi, %rdi, 2), %rax   // rax = rdi + 2*rdi = 3 * x
leaq (%rdi, %rdi, 4), %rax   // rax = rdi + 4*rdi = 5 * x
leaq 7(%rdi, %rsi),   %rax   // rax = rdi + rsi + 7 = x + y + 7
leaq (%rdi, %rdi, 8), %rax   // rax = rdi + 8*rdi = 9 * x
\end{codeexample}

\begin{important}
Unlike standard arithmetic instructions (\texttt{addq}, \texttt{subq}), the \texttt{leaq} instruction \textbf{never alters condition code flags}!
\end{important}

\begin{remark} \textbf{Subtleties of \texttt{subq} and \texttt{xorq}:}
\begin{itemize}
    \item \textbf{Operand Order in \texttt{subq}:} Remember AT\&T order: \texttt{subq \%rcx, \%rax} computes:
    $$\texttt{\%rax} \leftarrow \texttt{\%rax} - \texttt{\%rcx}$$
    The second operand is the minuend (the value being subtracted from).
    \item \textbf{Zeroing Registers:} The standard, idiomatically optimal way to zero a register is \texttt{xorl \%eax, \%eax}. This requires fewer instruction bytes than \texttt{movq \$0, \%rax} and is recognized by CPU decoders as a zero-latency register renaming operation.
\end{itemize}
\end{remark}

\subsubsection{Shift Operations}

\begin{definition} \textbf{Bit Shift Instructions:} 
Bit shifts shift the bits of destination $D$ by a shift count $k$:
\begin{itemize}
    \item \textbf{Left Shift (\texttt{salq} / \texttt{shlq}):} Shifts left, zero-filling the vacated lower bits. (Equivalent to multiplying by $2^k$).
    \item \textbf{Arithmetic Right Shift (\texttt{sarq}):} Shifts right, replicating the sign bit into vacated high-order bits. (Preserves two's complement sign for signed integer division by $2^k$).
    \item \textbf{Logical Right Shift (\texttt{shrq}):} Shifts right, zero-filling the vacated high-order bits. (Used for unsigned division by $2^k$).
\end{itemize}
\end{definition}

\begin{important} \textbf{The Shift Count Constraint:} 
The shift count operand can be an immediate constant (e.g., \texttt{salq \$3, \%rax}). However, if the shift count is stored in a register, x86-64 \textbf{strictly mandates} that it must reside in the single register \texttt{\%cl} (the low 8-bit byte of \texttt{\%rcx})!
No other register can be used as a variable shift counter. For 64-bit quadwords, only the lower 6 bits of \texttt{\%cl} are read ($0 \le k \le 63$).
\end{important}

\subsection{Control Flow: Condition Codes, Jumps, and Loops}

\subsubsection{Condition Code Flags}

\begin{definition} \textbf{Hardware Condition Codes (Status Flags):} 
The CPU maintains a set of single-bit status flags updated automatically by most arithmetic instructions:
\begin{itemize}
    \item \textbf{CF (Carry Flag):} Set if the most recent operation generated a carry out of the most significant bit. Used to detect \textbf{unsigned overflow}.
    \item \textbf{ZF (Zero Flag):} Set if the most recent operation yielded a result of \textbf{zero} (i.e., $\text{Result} == 0$).
    \item \textbf{SF (Sign Flag):} Set if the most recent operation yielded a \textbf{negative} result (the most significant bit is 1).
    \item \textbf{OF (Overflow Flag):} Set if the most recent operation caused a \textbf{signed two's-complement overflow} (e.g., adding two positive numbers yields a negative result, or subtracting a negative from positive overflows).
\end{itemize}
\end{definition}

\subsubsection{Comparison and Testing}

\begin{definition} \textbf{Comparison (\texttt{cmp}) and Test (\texttt{test}):} 
These instructions set condition flags without modifying any data registers:
\begin{itemize}
    \item \textbf{\texttt{cmpq S2, S1}:} Computes the difference $S_1 - S_2$ purely to set flags. It acts identically to \texttt{subq S2, S1}, except the difference is discarded:
    \begin{itemize}
        \item If $S_1 == S_2$: ZF is set.
        \item If $S_1 < S_2$ (signed): SF $\neq$ OF.
        \item If $S_1 < S_2$ (unsigned): CF is set.
    \end{itemize}
    \item \textbf{\texttt{testq S2, S1}:} Computes the bitwise conjunction $S_1 \ \& \ S_2$ purely to set flags. It acts identically to \texttt{andq S2, S1}, except the result is discarded:
    \begin{itemize}
        \item Most commonly used with identical operands: \texttt{testq \%rax, \%rax} sets ZF if \texttt{\%rax} is zero, and SF if \texttt{\%rax} is negative.
    \end{itemize}
\end{itemize}
\end{definition}

\subsubsection{Reading Condition Codes (\texttt{set} Instructions)}

\begin{definition} \textbf{The \texttt{set} Family:} 
Sets a single byte destination register (e.g., \texttt{\%al}) to 1 if the specified condition holds, and 0 otherwise:
\begin{itemize}
    \item \texttt{sete} / \texttt{setz} ($ZF=1$): Set on equal / zero ($S_1 == S_2$).
    \item \texttt{setne} / \texttt{setnz} ($ZF=0$): Set on not equal / non-zero ($S_1 \neq S_2$).
    \item \textbf{Signed Comparisons:}
    \begin{itemize}
        \item \texttt{setl} (less: $SF \oplus OF = 1$), \texttt{setle} (less or equal: $(SF \oplus OF) \mid ZF = 1$).
        \item \texttt{setg} (greater: $\sim(SF \oplus OF) \ \& \ \sim ZF = 1$), \texttt{setge} (greater or equal: $\sim(SF \oplus OF) = 1$).
    \end{itemize}
    \item \textbf{Unsigned Comparisons:}
    \begin{itemize}
        \item \texttt{setb} (below / carry: $CF=1$), \texttt{setbe} (below or equal: $CF \mid ZF = 1$).
        \item \texttt{seta} (above: $\sim CF \ \& \ \sim ZF = 1$), \texttt{setae} (above or equal: $\sim CF = 1$).
    \end{itemize}
\end{itemize}
\end{definition}

\subsubsection{Jumps and Branching}

\begin{definition} \textbf{Jump Instructions (\texttt{jmp} and \texttt{j*})}:
Alter execution by loading a new address into the program counter \texttt{\%rip}:
\begin{itemize}
    \item \textbf{Unconditional Jump (\texttt{jmp Target}):} Jumps directly to \texttt{Target}. Can be direct (\texttt{jmp .L2}) or indirect via register/memory (\texttt{jmp *\%rax}).
    \item \textbf{Conditional Jumps (\texttt{j* Target}):} Jump only if specific condition flags are set; otherwise, execution falls through to the next sequential instruction.
\end{itemize}
\end{definition}

\begin{codeexample}
// C code:
long max(long a, long b) {
    if (a > b) return a;
    else return b;
}

// Corresponding x86-64 assembly:
max:
    cmpq %rsi, %rdi      // Compare a (%rdi) and b (%rsi) by computing rdi - rsi
    jle  .L_b_is_larger  // If a <= b, jump to .L_b_is_larger
    movq %rdi, %rax      // Otherwise, return a (%rdi)
    ret
.L_b_is_larger:
    movq %rsi, %rax      // Return b (%rsi)
    ret
\end{codeexample}

\subsubsection{Conditional Moves (\texttt{cmov})}

\begin{definition} \textbf{Conditional Move (\texttt{cmov}):} 
Copies data from source to destination only if condition flags satisfy the specified predicate:
$$\texttt{cmov}\langle\text{condition}\rangle \quad \text{Source}, \quad \text{Destination}$$
If the condition is false, the instruction acts as a no-op (\texttt{nop}).
\end{definition}

\begin{remark} \textbf{Intuition \& Branch Prediction Optimization:} 
Modern high-performance processors rely on deep pipelining (15--20 pipeline stages). To keep the pipeline full, the processor employs a \textbf{branch predictor} to speculate which way a conditional jump will go before the condition is evaluated.
\begin{itemize}
    \item When a branch is predicted correctly, execution proceeds without delay.
    \item When a branch is \textbf{mispredicted}, the entire instruction pipeline must be flushed, throwing away partially executed instructions and costing \textbf{15--30 wasted clock cycles}.
\end{itemize}
A \textbf{conditional move (\texttt{cmov})} computes both potential outcomes and then conditionally selects the result using data flow rather than control flow jumps. Because there are no jump instructions, there is \textbf{zero branch misprediction penalty}!
\end{remark}

\begin{codeexample}
// Optimizing max() with conditional move:
max:
    movq %rsi, %rax      // rax = b
    cmpq %rsi, %rdi      // Compare a (%rdi) and b (%rsi)
    cmovg %rdi, %rax     // If a > b, overwrite rax with a; else keep b
    ret                  // Done! No branching, no pipeline stall!
\end{codeexample}

\subsection{Procedures and the Calling Convention (System V ABI)}

\begin{definition} \textbf{The System V AMD64 Calling Convention:} 
The standard Application Binary Interface (ABI) used on Linux and macOS defining how compiled functions interact, pass parameters, return values, and manage registers.
\end{definition}

\subsubsection{Argument Passing (The 6-Register Rule)}

\begin{theorem} \textbf{Passing Function Arguments in Registers:} 
Under the System V ABI, the first 6 integer or pointer arguments to a function are passed directly inside registers in strict order:
\begin{enumerate}
    \item 1st Argument: \texttt{\%rdi}
    \item 2nd Argument: \texttt{\%rsi}
    \item 3rd Argument: \texttt{\%rdx}
    \item 4th Argument: \texttt{\%rcx}
    \item 5th Argument: \texttt{\%r8}
    \item 6th Argument: \texttt{\%r9}
\end{enumerate}
Any additional arguments (7th, 8th, etc.) are pushed onto the runtime stack in reverse order. The primary return value is returned in \texttt{\%rax} (and \texttt{\%rdx} for 128-bit results).
\end{theorem}

\begin{remark} \textbf{Mnemonic to Remember Arguments:} 
A famous mnemonic to memorize the argument register sequence is:
$$\textbf{rDI}ane's \quad \textbf{rSI}lk \quad \textbf{rDX}ress \quad \textbf{rCX}ost \quad \textbf{r8}9 \quad (\texttt{\%rdi}, \texttt{\%rsi}, \texttt{\%rdx}, \texttt{\%rcx}, \texttt{\%r8}, \texttt{\%r9})$$
\end{remark}

\subsubsection{Caller-Saved vs. Callee-Saved Registers}

\begin{definition} \textbf{Register Preservation Roles:} 
To ensure functions do not corrupt each other's local state, registers are strictly divided into two categories:
\begin{itemize}
    \item \textbf{Caller-Saved Registers (Call-Clobbered):}
    $$\texttt{\%rax}, \quad \texttt{\%rdi}, \quad \texttt{\%rsi}, \quad \texttt{\%rdx}, \quad \texttt{\%rcx}, \quad \texttt{\%r8}, \quad \texttt{\%r9}, \quad \texttt{\%r10}, \quad \texttt{\%r11}$$
    These registers can be freely modified by the called function (\emph{callee}). If the calling function (\emph{caller}) needs their values preserved across a \texttt{call}, the caller must save them on the stack prior to making the call.
    
    \item \textbf{Callee-Saved Registers (Preserved Registers):}
    $$\texttt{\%rbx}, \quad \texttt{\%rbp}, \quad \texttt{\%r12}, \quad \texttt{\%r13}, \quad \texttt{\%r14}, \quad \texttt{\%r15}, \quad \texttt{\%rsp}$$
    The called function (\emph{callee}) is strictly obligated to preserve these registers. If the callee intends to use any of them, it \textbf{must save their original values} (typically via \texttt{pushq}) upon entry, and \textbf{restore them} (via \texttt{popq}) before returning with \texttt{ret}.
\end{itemize}
\end{definition}

\subsubsection{Procedure Control: \texttt{call} and \texttt{ret}}

\begin{definition} \textbf{Mechanics of \texttt{call} and \texttt{ret}:} 
Functions transfer control through the stack:
\begin{itemize}
    \item \textbf{\texttt{call Target}:}
    \begin{enumerate}
        \item Pushes the 8-byte \textbf{return address} (the memory address of the instruction immediately following the \texttt{call}) onto the stack:
        $$\texttt{\%rsp} \leftarrow \texttt{\%rsp} - 8, \quad \text{Mem}[\texttt{\%rsp}] \leftarrow \texttt{\%rip}_{\text{next}}$$
        \item Jumps to the target function: $\texttt{\%rip} \leftarrow \text{Target}$.
    \end{enumerate}
    \item \textbf{\texttt{ret}:}
    \begin{enumerate}
        \item Pops the return address off the top of the stack:
        $$\texttt{\%rip} \leftarrow \text{Mem}[\texttt{\%rsp}], \quad \texttt{\%rsp} \leftarrow \texttt{\%rsp} + 8$$
        \item Resumes execution at the retrieved address.
    \end{enumerate}
\end{itemize}
\end{definition}

\begin{important} \textbf{The 16-Byte Stack Alignment Requirement:} 
The System V AMD64 ABI requires that the stack pointer \texttt{\%rsp} must be a multiple of 16 (16-byte aligned) immediately before any \texttt{call} instruction is issued. 
When a \texttt{call} executes, it pushes the 8-byte return address, leaving \texttt{\%rsp} offset by 8 upon function entry. Functions must ensure their stack allocations preserve this 16-byte alignment before calling any sub-functions!
\end{important}

\subsection{Stack Frames, Local Storage, and Buffer Overflows}

\begin{definition} \textbf{Stack Frame Anatomy:} 
Each function invocation allocates a contiguous region of memory on the stack called a \textbf{stack frame}:
\begin{itemize}
    \item \textbf{Top of Frame (High Address):} 7th+ incoming arguments passed by caller, followed by the 8-byte Return Address.
    \item \textbf{Saved Frame Pointer (\texttt{\%rbp}):} (Optional) Preserved base pointer of the caller.
    \item \textbf{Saved Callee-Saved Registers:} Registers preserved by the function.
    \item \textbf{Local Variables:} Local arrays or structures that cannot fit entirely inside registers.
    \item \textbf{Outgoing Arguments (Low Address / \texttt{\%rsp}):} Space for arguments passed to called subroutines.
\end{itemize}
\end{definition}

\begin{codeexample}
// Standard Function Prologue and Epilogue:
function_example:
    pushq %rbp              // Save caller's base pointer
    movq %rsp, %rbp         // Set up new base pointer for current frame
    subq $32, %rsp          // Allocate 32 bytes for local stack variables
    
    // ... Function Body ...
    
    movq %rbp, %rsp         // Deallocate local storage (or use 'leave')
    popq %rbp               // Restore caller's base pointer
    ret                     // Pop return address and return to caller
\end{codeexample}

\begin{definition} \textbf{Buffer Overflow (Stack Smashing):} 
Occurs when a program writes data beyond the allocated boundary of a stack buffer (e.g., using unsafe functions like \texttt{gets()} or \texttt{strcpy()}). 
Because the runtime stack grows downward, local buffers are located below the saved return address. Overflowing the buffer writes upwards, overwriting the saved frame pointer and the \textbf{saved return address}. When \texttt{ret} executes, the CPU pops the attacker-controlled address into \texttt{\%rip}, hijacking program control flow.
\end{definition}

\begin{definition} \textbf{Modern Stack Protection Defenses:}
\begin{itemize}
    \item \textbf{Stack Canaries (\texttt{-fstack-protector}):} The compiler inserts a random secret value (canary) from \texttt{\%fs:40} between local stack buffers and the saved return address. Before executing \texttt{ret}, the function compares the canary against \texttt{\%fs:40}. If a buffer overflow altered the canary, the program aborts with a stack smashing crash rather than executing attacker code.
    \item \textbf{No-Execute (NX / W\^{}X Bit):} Memory pages containing the stack are marked as read-write but strictly non-executable, preventing injected shellcode on the stack from executing.
    \item \textbf{Address Space Layout Randomization (ASLR):} Randomizes the base addresses of the stack, heap, and shared libraries upon every execution, making it difficult for attackers to predict fixed code injection targets.
\end{itemize}
\end{definition}
