\section{Schutz fremder und eigener Inhalte}

\subsection{Einführung und Beispiele}

\begin{example}
    \textbf{Aktuelle Fragestellungen und Beispiele:}
    \begin{itemize}
        \item Affen-Selfie (Wem gehören die Rechte?)
        \item KI-Training mit geschützten Daten
        \item KI ersetzt Online-Informationsdienste mit deren eigenen Inhalten
        \item Gibt es ein Opt-Out-Recht für KI-Training?
    \end{itemize}
\end{example}

\begin{remark}
    Verschiedene Gesetze regeln den Schutz von Inhalten und weisen
    unterschiedliche Ansätze auf. Relevante Rechtsgebiete sind unter anderem:
    \begin{itemize}
        \item Urheberrechtsgesetz
        \item Gesetz gegen den unlauteren Wettbewerb
        \item Obligationenrecht
        \item Markenschutz
        \item Datenschutz
    \end{itemize}
\end{remark}

\subsection{Das Urheberrecht}

\begin{definition}
    \textbf{Urheberrecht:} Das Urheberrecht schützt eine schöpferische Leistung.
    Der Urheber hat das Recht, über die Verwendung dieser Leistung zu bestimmen.
\end{definition}

\begin{remark}
    \textbf{Was schützt das Urheberrecht?} \\
    Der Weg von der Idee zur Verkörperung: Idee / Gedanke / Information
    $\rightarrow$ Formulierung / Umsetzung $\rightarrow$ Verkörperlichung. \\
    Das Urheberrecht schützt die \textbf{Formulierung bzw. Umsetzung}. \\
    \textit{"Schutzfähig ist nur der bestimmte Ausdruck eines Gedankens. Erst
    seine konkrete Verkörperung macht das Werk aus."}
\end{remark}

\begin{definition}
    \textbf{Bedingungen für urheberrechtlichen Schutz:} Damit ein Werk geschützt
    ist, müssen grundsätzlich folgende Bedingungen erfüllt sein:
    \begin{itemize}
        \item \textbf{Geistige Schöpfung:} Es muss ein Gestaltungswille bzw.
              eine bewusste Kreation vorliegen.
        \item \textbf{Individueller Charakter:} Das Werk muss sich abheben und
              kreativ sein.
    \end{itemize}
\end{definition}

\begin{remark}
    \textbf{Neuerung 2020 (Lichtbildschutz):} Gemäss dem neuen Gesetz von 2020
    benötigen Fotografien (z.B. von dreidimensionalen Objekten) keinen
    individuellen Charakter mehr, um rechtlich geschützt zu sein.
\end{remark}

\subsection{Verwertungs- und Persönlichkeitsrechte}

\begin{definition}
    \textbf{Urheberpersönlichkeitsrechte:} Schützen die persönliche Beziehung
    des Urhebers zum Werk. Dazu gehören das Recht auf Autorennennung, das Recht
    über die Erstveröffentlichung zu entscheiden sowie der Schutz vor
    Entstellung oder Beeinträchtigung des Werkes.
\end{definition}

\begin{remark}
    \textbf{Verwertungsrechte (Art. 10 \& 11 URG):} Der Urheber bestimmt allein,
    ob, wann und wie das Werk verwendet wird.
    \begin{itemize}
        \item Vervielfältigungs-, Verbreitungs-, Vermiet-, Vortrags-, Vorführ-,
              Sende- und Wiedergaberecht.
        \item Bearbeitungsrecht (z.B. Softwarewartung).
        \item Generalklausel erfasst auch neuartige Nutzungen (z.B. Framing im
              Web).
        \item Schutzdauer: 70 Jahre nach dem Tod des Urhebers (50 Jahre bei
              Computerprogrammen und Fotos ohne individuellen Charakter).
    \end{itemize}
\end{remark}

\begin{definition}
    \textbf{Verwandte Schutzrechte:} Schutzrechte für Leistungen, die nicht
    zwingend eigene Werke darstellen:
    \begin{itemize}
        \item \textbf{Ausübende Künstler (Interpreten):} Recht auf
              Wahrnehmbarmachung, Aufnahme, Vervielfältigung und Verbreitung
              ihrer Darbietung.
        \item \textbf{Hersteller von Ton- und Tonbildträgern:} Recht auf
              Vervielfältigung und Verbreitung der Aufnahmen.
        \item \textbf{Sendeunternehmen:} Recht auf Weitersendung und Aufnahme
              ihrer Sendungen.
    \end{itemize}
\end{definition}

\subsection{Beschaffung von Rechten und Lizenzierung}

\begin{definition}
    \textbf{Lizenzierung vs. Übertragung:}
    \begin{itemize}
        \item \textbf{Lizenzierung:} Einräumung der Befugnis, das Werk auf
              bestimmte Weise zu nutzen. Der Rechteinhaber behält seine Rechte.
        \item \textbf{Übertragung (Art. 16 URG):} Vollständige Abtretung der
              vermögensrechtlichen Befugnisse.
    \end{itemize}
\end{definition}

\begin{remark}
    \textbf{Urheberrecht im Arbeits- und Auftragsverhältnis:}
    \begin{itemize}
        \item Rechte gehen nicht automatisch auf den Arbeitgeber über.
        \item \textbf{Ausnahme (Art. 17 URG):} Computerprogramme, die
              Arbeitnehmer in Ausübung dienstlicher Pflichten schaffen, stehen
              dem Arbeitgeber zu.
        \item Bei Kunden/Aufträgen bestimmt sich der Umfang der Nutzungsrechte
              nach dem Vertragszweck (Zweckübertragungstheorie; im Zweifel
              verbleiben Rechte beim Urheber).
    \end{itemize}
\end{remark}

\begin{remark}
    \textbf{Verwertungsgesellschaften (Art. 40 ff. URG):} Staatlich zugelassene
    Organisationen (z.B. ProLitteris, SUISA, Suissimage) zur kollektiven
    Rechteverwertung im Monopolbereich (z.B. Leerträgervergütung,
    Kabelweitersendung) und Marktbereich.
\end{remark}

\subsection{Haftung von Plattformen und Hosting-Diensten}

\begin{remark}
    \textbf{Hosting-Provider-Pflichten (Art. 39d URG \& EU-Recht):}
    \begin{itemize}
        \item Pflicht zur Verhinderung der erneuten rechtswidrigen
              Zugänglichmachung ("Notice and Takedown" / "Stay down"), wenn der
              Dienst eine besondere Gefahr für Verletzungen geschaffen hat.
        \item \textbf{EU-Urheberrechtsreform (Art. 17):} Plattformen haften für
              Urheberrechtsverletzungen ihrer Nutzer, ausser sie weisen nach,
              dass sie ausreichende Bemühungen für Lizenzen und technische
              Schutzmassnahmen (Upload-Filter) unternommen haben.
    \end{itemize}
\end{remark}

\subsection{Gesetzliche Schranken und Digital Rights Management}

\begin{definition}
    \textbf{Schranken des Urheberrechts:} Gesetzlich erlaubte Verwendungen
    geschützter Werke ohne vorherige Zustimmung des Urhebers:
    \begin{itemize}
        \item \textbf{Eigengebrauch (Art. 19 URG):} Nutzung im persönlichen
              Kreis, durch Lehrpersonen im Unterricht oder betriebsintern zur
              Information/Dokumentation. Download zum Eigengebrauch ist in der
              Schweiz legal, Upload/Anbieten jedoch rechtswidrig.
        \item \textbf{Zitatrecht (Art. 25 URG):} Veröffentlichte Werke dürfen
              zur Erläuterung/Veranschaulichung im erforderlichen Umfang mit
              Quellenangabe zitiert werden.
        \item \textbf{Parodie \& Karikatur:} Abwandlungen eines Werks zur
              humorvollen oder kritischen Auseinandersetzung sind zulässig.
        \item \textbf{Erschöpfungsgrundsatz (Art. 12 URG):} Wurde ein
              Werkexemplar veräusserungsgemäss verkauft, darf es weiterverkauft
              werden (gilt auch für Online-Kopien von Software).
        \item \textbf{Weitere Schranken:} Berichterstattung über aktuelle
              Ereignisse, Wissenschaftsschranke (Art. 24d URG) und
              vorübergehende Vervielfältigungen (Art. 24a URG).
    \end{itemize}
\end{definition}

\begin{remark}
    \textbf{Digital Rights Management (DRM, Art. 39a \& 39c URG):}
    \begin{itemize}
        \item Das Umgehen wirksamer technischer Schutzmassnahmen ist verboten.
        \item Elektronische Informationen zur Rechteverwaltung (z.B. digitale
              Wasserzeichen, Copyright- Vermerke) dürfen weder entfernt noch
              verändert werden.
    \end{itemize}
\end{remark}

\subsection{KI und Urheberrecht}

\begin{remark}
    \textbf{Training von KI-Sprachmodellen (LLMs):}
    \begin{itemize}
        \item Das Training dient in der Regel nicht dem "menschlichen Genuss"
              des Werks und führt bei hinreichender Zerlegung/Transformation
              nicht zu einer klassischen Kopie im Modell (Werk verblasst).
        \item Mögliche Schranken: Betrieblicher Eigengebrauch (Art. 19 URG) oder
              Wissenschaftsschranke (Art. 24d URG), sofern kein Re-Memorisieren
              stattfindet.
        \item \textbf{KI-Outputs:} KI-generierter Content ist nur
              urheberrechtlich geschützt, wenn der menschliche Prompt dem Output
              eine ausreichend individuelle Prägung verleiht.
    \end{itemize}
\end{remark}

\subsection{Rechtsfolgen und Schutz ausserhalb des URG}

\begin{remark}
    \textbf{Rechtsfolgen bei Urheberrechtsverletzungen:}
    \begin{itemize}
        \item \textbf{Zivilrecht:} Schadenersatz (nach nachgewiesenem Schaden
              oder Lizenzanalogie) sowie Gewinnherausgabe.
        \item \textbf{Strafrecht (Art. 67 ff. URG):} Vorsätzliche Verletzungen
              werden mit Geld- oder Freiheitsstrafe geahndet (inkl.
              Eventualvorsatz und Gehilfenschaft).
    \end{itemize}
\end{remark}

\begin{remark}
    \textbf{Schutz ausserhalb des Urheberrechts:}
    \begin{itemize}
        \item \textbf{Lauterkeitsrecht (Art. 5 lit. c UWG):} Unlauter handelt,
              wer das marktreife Arbeitsergebnis eines anderen ohne angemessenen
              eigenen Aufwand durch technische Reproduktionsverfahren (z.B.
              Screen Scraping von Inseraten) übernimmt und verwertet.
        \item \textbf{Nutzungsbedingungen (AGB):} Binden Besucher von Webseiten
              grundsätzlich nur, wenn eine gültige Vertragsbeziehung zustande
              gekommen ist (z.B. durch Registrierung).
    \end{itemize}
\end{remark}

\subsection{Domain-Namen}

\subsubsection{Grundlagen und Aufbau}

\begin{definition}
    \textbf{Domain-Name:} Weltweit eindeutige, menschenlesbare Adresse zur Identifikation und Adressierung von Rechnern und Diensten im Internet (wird über das DNS in IP-Adressen aufgelöst).
\end{definition}

\begin{remark}
    \textbf{Hierarchischer Aufbau:}
    \begin{itemize}
        \item \textbf{Top-Level-Domain (TLD):} Oberste Ebene der Namenshierarchie:
              \begin{itemize}
                  \item \textit{ccTLD (country code):} Länderspezifisch (z.B. \texttt{.ch}, \texttt{.de})
                  \item \textit{gTLD (generic):} Generisch (z.B. \texttt{.com}, \texttt{.org}, \texttt{.swiss})
              \end{itemize}
        \item \textbf{Second-Level-Domain (SLD):} Der eigentliche gewählte Name (z.B. \texttt{ethz} in \texttt{ethz.ch}).
        \item \textbf{Subdomains:} Weitere Untergliederungen (z.B. \texttt{moodle.ethz.ch}).
    \end{itemize}
\end{remark}

\subsubsection{Vergabesystem und Akteure}

\begin{remark}
    \textbf{Akteure bei der Domain-Vergabe:}
    \begin{itemize}
        \item \textbf{ICANN:} Globale Non-Profit-Organisation, koordiniert das Domain Name System und die Root-Zone weltweit.
        \item \textbf{Nationale Ordnung (Registry):} Verwaltet die Datenbank der jeweiligen TLD (z.B. SWITCH bzw. BAKOM für \texttt{.ch}).
        \item \textbf{Händler (Registrar):} Akkreditierte Verkäufer (z.B. Hostpoint, Infomaniak), über die Endkunden Domains registrieren.
        \item \textbf{Registrant:} Der Endkunde bzw. Domain-Halter.
    \end{itemize}
\end{remark}

\subsubsection{Rechtliche Natur und Vergabe}

\begin{definition}
    \textbf{First come, first served:} Domain-Namen werden nach dem Prioritätsprinzip vergeben: Wer eine freie Domain zuerst anmeldet, erhält den Zuschlag. Die Registrierungsstellen führen keine Vorprüfung auf allfällige Marken- oder Namensrechte durch.
\end{definition}

\begin{important}
    \textbf{Dienstleistung, nicht Eigentum!} \\
    An einem Domain-Namen erwirbt man \textbf{kein Sacheigentum} (kein Eigentum im Sinne von Art. 641 ZGB). \\
    Es handelt sich um eine rein \textbf{vertragliche Dienstleistung}: Ein schuldrechtliches, relatives Nutzungsrecht auf Eintragung und Routing im DNS gegen Bezahlung einer periodischen Gebühr.
\end{important}

\subsubsection{Kennzeichenfunktion und Schutzrechte}

\begin{definition}
    \textbf{Kennzeichenfunktion von Domain-Namen:} Obwohl Domain-Namen technisch primär der Adressierung dienen, fungieren sie im geschäftlichen Verkehr als Erkennungs- und Unterscheidungszeichen. Sie wirken faktisch wie Namen oder Kennzeichen. Daher muss bei der Registrierung und Nutzung ein \textbf{hinreichender Abstand} zu bereits bestehenden Schutzrechten Dritter gewahrt werden, um Verwechslungsgefahr zu vermeiden.
\end{definition}

\begin{remark}
    \textbf{Anwendbare Rechtsgrundlagen bei Konflikten:}
    \begin{itemize}
        \item \textbf{Namensschutz (Art. 29 ZGB):} Schützt den Namen von natürlichen und juristischen Personen sowie Gemeinwesen vor unbefugter Namensanmassung. Gilt \textbf{branchenunabhängig}.
        \item \textbf{Markenschutz (MSchG):} Schützt eingetragene Kennzeichen für bestimmte Waren oder Dienstleistungen. Gilt grundsätzlich \textbf{branchenabhängig} (Spezialitätsprinzip: Schutz nur für die eingetragenen Klassen; Ausnahme: berühmte Marken gemäss Art. 15 MSchG).
        \item \textbf{Lauterkeitsrecht (Art. 3 Abs. 1 lit. d UWG):} Verbietet unlautere Verhaltensweisen, die geeignet sind, Verwechslungen mit den Waren, Werken, Leistungen oder dem Geschäftsbetrieb eines anderen herbeizuführen (sowie unlautere Kundenabfangung oder Rufausbeutung).
        \item \textbf{Firmenrecht (Art. 956 OR):} Schützt die im Handelsregister eingetragene Geschäftsfirma vor unbefugtem Gebrauch durch Dritte.
    \end{itemize}
\end{remark}

\subsubsection{Rechtsverletzung und Verwechslungsgefahr}

\begin{remark}
    \textbf{Rechtsverletzung durch Fehlleitung (Initial Interest Confusion):} \\
    Eine Rechtsverletzung (insbesondere nach UWG oder Kennzeichenrecht) kann bereits dann vorliegen, wenn Suchende oder Kunden, die gezielt nach dem berechtigten Original suchen, durch den Domain-Namen auf eine falsche Website fehlgeleitet werden (\textbf{Kundenabfangung}). \\
    Unerheblich ist dabei grundsätzlich, ob der Internetnutzer beim Betrachten des Website-Inhalts bemerkt, dass es sich nicht um die gesuchte Originalseite handelt, da dessen Aufmerksamkeit bereits unlauter abgelenkt wurde.
\end{remark}

\subsubsection{Sonderkonstellationen und Schranken}

\begin{remark}
    \textbf{Spezialfälle im Domainrecht:}
    \begin{itemize}
        \item \textbf{Gemeindenamen und Hoheitszeichen (z.B. \texttt{gemeindename.ch}):} Namen von Gemeinwesen (Kantone, Gemeinden, Städte) geniessen Namensschutz (Art. 29 ZGB). Private dürfen sich Gemeindenamen in der Regel nicht als Domain registrieren, da das Publikum dahinter eine offizielle behördliche Website vermutet (\textbf{hohe Verkehrserwartung}).
        \item \textbf{Beschreibende Begriffe und Gemeingut:} Rein beschreibende Gattungsbezeichnungen (z.B. \texttt{velo.ch}, \texttt{recht.ch}) gehören zum Gemeingut (Freihaltebedürfnis) und geniessen mangels Unterscheidungskraft in der Regel keinen Markenschutz. Bei reinen Sachbezeichnungen gilt strikt das Prioritätsprinzip (\textit{First come, first served}), sofern keine unlautere Behinderung oder Täuschung vorliegt.
        \item \textbf{Kollision gleichwertiger Rechte (Gleichnamige):} Haben mehrere Parteien ein legitimes Recht am gleichen Namen bzw. Zeichen (z.B. gleiche Familiennamen oder Firmen in unterschiedlichen Branchen):
              \begin{itemize}
                  \item \textit{Grundsatz:} Es gilt das Prioritätsprinzip (\textit{First come, first served}).
                  \item \textit{Korrektiv:} Sofern eine Partei eine deutlich überwiegende \textbf{Verkehrserwartung} für sich beanspruchen kann (z.B. überragende Bekanntheit) oder die Registrierung rechtsmissbräuchlich bzw. sittenwidrig erfolgte, kann ein Vorrang bestehen.
              \end{itemize}
        \item \textbf{Nutzung durch Händler, Vertriebspartner und Importeure:} Dritte dürfen fremde Marken nutzen, um auf angebotene Originalwaren, Ersatzteile oder Dienstleistungen wahrheitsgemäss hinzuweisen (beschreibender Markengebrauch / Erschöpfungsgrundsatz). Es darf jedoch nicht der irreführende Eindruck erweckt werden, man sei die Markeninhaberin selbst oder eine wirtschaftlich verbundene Niederlassung.
    \end{itemize}
\end{remark}

\subsubsection{Streitbeilegung und Schlichtungsverfahren}

\begin{definition}
    \textbf{Alternative Streitbeilegung (Dispute Resolution):} Schnelle, kostengünstige und spezialisierte Verfahren zur Beilegung von Domain-Streitigkeiten ausserhalb der ordentlichen staatlichen Zivilgerichte.
\end{definition}

\begin{remark}
    \textbf{Verfahren im Überblick:}
    \begin{itemize}
        \item \textbf{UDRP (Uniform Domain-Name Dispute-Resolution Policy):}
              \begin{itemize}
                  \item Globales Schlichtungsverfahren der ICANN für generische Top-Level-Domains (gTLDs wie \texttt{.com}, \texttt{.org}, \texttt{.net}) sowie diverse ccTLDs.
                  \item Streitigkeiten werden massgeblich über das Schiedszentrum der \textbf{WIPO} (World Intellectual Property Organization in Genf) geführt.
                  \item \textit{Voraussetzungen (kumulativ):} Die Domain ist identisch oder verwechslungsfähig ähnlich zu einer Marke des Klägers, der Halter hat keine eigenen Rechte oder berechtigten Interessen und die Domain wurde bösgläubig registriert und wird bösgläubig genutzt (\textit{bad faith} / Cybersquatting / Domaingrabbing).
              \end{itemize}
        \item \textbf{SWITCH-Streitbeilegungsverfahren (für \texttt{.ch} und \texttt{.li}):}
              \begin{itemize}
                  \item Spezifisches Verfahren für Schweizer und Liechtensteiner Domains, ebenfalls unter administrativer Leitung der \textbf{WIPO}.
                  \item Zweistufiges Verfahren: Zunächst eine Schlichtungsverhandlung (Mediation); falls keine Einigung erzielt wird, entscheidet ein unabhängiger Fachexperte.
                  \item \textit{Voraussetzungen:} Die Zuteilung oder Verwendung der Domain stellt eine Verletzung eines Kennzeichenrechts nach schweizerischem Recht dar (Marken-, Namens-, Firmenrecht oder UWG) und der Inhaber besitzt kein überwiegendes schutzwürdiges Eigeninteresse.
              \end{itemize}
    \end{itemize}
\end{remark}
