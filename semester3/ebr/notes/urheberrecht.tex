\section{Schutz fremder und eigener Inhalte}

\subsection{Einführung und Beispiele}

\begin{example}
    \textbf{Aktuelle Fragestellungen und Beispiele:}
    \begin{itemize}
        \item Affen-Selfie (Wem gehören die Rechte?)
        \item KI-Training mit geschützten Daten
        \item KI ersetzt Online-Informationsdienste mit deren eigenen Inhalten
        \item Gibt es ein Opt-Out-Recht für KI-Training?
    \end{itemize}
\end{example}

\begin{remark}
    Verschiedene Gesetze regeln den Schutz von Inhalten und weisen
    unterschiedliche Ansätze auf. Relevante Rechtsgebiete sind unter anderem:
    \begin{itemize}
        \item Urheberrechtsgesetz
        \item Gesetz gegen den unlauteren Wettbewerb
        \item Obligationenrecht
        \item Markenschutz
        \item Datenschutz
    \end{itemize}
\end{remark}

\subsection{Das Urheberrecht}

\begin{definition}
    \textbf{Urheberrecht:} Das Urheberrecht schützt eine schöpferische Leistung.
    Der Urheber hat das Recht, über die Verwendung dieser Leistung zu bestimmen.
\end{definition}

\begin{remark}
    \textbf{Was schützt das Urheberrecht?} \\
    Der Weg von der Idee zur Verkörperung: Idee / Gedanke / Information
    $\rightarrow$ Formulierung / Umsetzung $\rightarrow$ Verkörperlichung. \\
    Das Urheberrecht schützt die \textbf{Formulierung bzw. Umsetzung}. \\
    \textit{"Schutzfähig ist nur der bestimmte Ausdruck eines Gedankens. Erst
    seine konkrete Verkörperung macht das Werk aus."}
\end{remark}

\begin{definition}
    \textbf{Bedingungen für urheberrechtlichen Schutz:} Damit ein Werk geschützt
    ist, müssen grundsätzlich folgende Bedingungen erfüllt sein:
    \begin{itemize}
        \item \textbf{Geistige Schöpfung:} Es muss ein Gestaltungswille bzw.
              eine bewusste Kreation vorliegen.
        \item \textbf{Individueller Charakter:} Das Werk muss sich abheben und
              kreativ sein.
    \end{itemize}
\end{definition}

\begin{remark}
    \textbf{Neuerung 2020 (Lichtbildschutz):} Gemäss dem neuen Gesetz von 2020
    benötigen Fotografien (z.B. von dreidimensionalen Objekten) keinen
    individuellen Charakter mehr, um rechtlich geschützt zu sein.
\end{remark}

\subsection{Verwertungs- und Persönlichkeitsrechte}

\begin{definition}
    \textbf{Urheberpersönlichkeitsrechte:} Schützen die persönliche Beziehung
    des Urhebers zum Werk. Dazu gehören das Recht auf Autorennennung, das Recht
    über die Erstveröffentlichung zu entscheiden sowie der Schutz vor
    Entstellung oder Beeinträchtigung des Werkes.
\end{definition}

\begin{remark}
    \textbf{Verwertungsrechte (Art. 10 \& 11 URG):} Der Urheber bestimmt allein,
    ob, wann und wie das Werk verwendet wird.
    \begin{itemize}
        \item Vervielfältigungs-, Verbreitungs-, Vermiet-, Vortrags-, Vorführ-,
              Sende- und Wiedergaberecht.
        \item Bearbeitungsrecht (z.B. Softwarewartung).
        \item Generalklausel erfasst auch neuartige Nutzungen (z.B. Framing im
              Web).
        \item Schutzdauer: 70 Jahre nach dem Tod des Urhebers (50 Jahre bei
              Computerprogrammen und Fotos ohne individuellen Charakter).
    \end{itemize}
\end{remark}

\begin{definition}
    \textbf{Verwandte Schutzrechte:} Schutzrechte für Leistungen, die nicht
    zwingend eigene Werke darstellen:
    \begin{itemize}
        \item \textbf{Ausübende Künstler (Interpreten):} Recht auf
              Wahrnehmbarmachung, Aufnahme, Vervielfältigung und Verbreitung
              ihrer Darbietung.
        \item \textbf{Hersteller von Ton- und Tonbildträgern:} Recht auf
              Vervielfältigung und Verbreitung der Aufnahmen.
        \item \textbf{Sendeunternehmen:} Recht auf Weitersendung und Aufnahme
              ihrer Sendungen.
    \end{itemize}
\end{definition}

\subsection{Beschaffung von Rechten und Lizenzierung}

\begin{definition}
    \textbf{Lizenzierung vs. Übertragung:}
    \begin{itemize}
        \item \textbf{Lizenzierung:} Einräumung der Befugnis, das Werk auf
              bestimmte Weise zu nutzen. Der Rechteinhaber behält seine Rechte.
        \item \textbf{Übertragung (Art. 16 URG):} Vollständige Abtretung der
              vermögensrechtlichen Befugnisse.
    \end{itemize}
\end{definition}

\begin{remark}
    \textbf{Urheberrecht im Arbeits- und Auftragsverhältnis:}
    \begin{itemize}
        \item Rechte gehen nicht automatisch auf den Arbeitgeber über.
        \item \textbf{Ausnahme (Art. 17 URG):} Computerprogramme, die
              Arbeitnehmer in Ausübung dienstlicher Pflichten schaffen, stehen
              dem Arbeitgeber zu.
        \item Bei Kunden/Aufträgen bestimmt sich der Umfang der Nutzungsrechte
              nach dem Vertragszweck (Zweckübertragungstheorie; im Zweifel
              verbleiben Rechte beim Urheber).
    \end{itemize}
\end{remark}

\begin{remark}
    \textbf{Verwertungsgesellschaften (Art. 40 ff. URG):} Staatlich zugelassene
    Organisationen (z.B. ProLitteris, SUISA, Suissimage) zur kollektiven
    Rechteverwertung im Monopolbereich (z.B. Leerträgervergütung,
    Kabelweitersendung) und Marktbereich.
\end{remark}

\subsection{Haftung von Plattformen und Hosting-Diensten}

\begin{remark}
    \textbf{Hosting-Provider-Pflichten (Art. 39d URG \& EU-Recht):}
    \begin{itemize}
        \item Pflicht zur Verhinderung der erneuten rechtswidrigen
              Zugänglichmachung ("Notice and Takedown" / "Stay down"), wenn der
              Dienst eine besondere Gefahr für Verletzungen geschaffen hat.
        \item \textbf{EU-Urheberrechtsreform (Art. 17):} Plattformen haften für
              Urheberrechtsverletzungen ihrer Nutzer, ausser sie weisen nach,
              dass sie ausreichende Bemühungen für Lizenzen und technische
              Schutzmassnahmen (Upload-Filter) unternommen haben.
    \end{itemize}
\end{remark}

\subsection{Gesetzliche Schranken und Digital Rights Management}

\begin{definition}
    \textbf{Schranken des Urheberrechts:} Gesetzlich erlaubte Verwendungen
    geschützter Werke ohne vorherige Zustimmung des Urhebers:
    \begin{itemize}
        \item \textbf{Eigengebrauch (Art. 19 URG):} Nutzung im persönlichen
              Kreis, durch Lehrpersonen im Unterricht oder betriebsintern zur
              Information/Dokumentation. Download zum Eigengebrauch ist in der
              Schweiz legal, Upload/Anbieten jedoch rechtswidrig.
        \item \textbf{Zitatrecht (Art. 25 URG):} Veröffentlichte Werke dürfen
              zur Erläuterung/Veranschaulichung im erforderlichen Umfang mit
              Quellenangabe zitiert werden.
        \item \textbf{Parodie \& Karikatur:} Abwandlungen eines Werks zur
              humorvollen oder kritischen Auseinandersetzung sind zulässig.
        \item \textbf{Erschöpfungsgrundsatz (Art. 12 URG):} Wurde ein
              Werkexemplar veräusserungsgemäss verkauft, darf es weiterverkauft
              werden (gilt auch für Online-Kopien von Software).
        \item \textbf{Weitere Schranken:} Berichterstattung über aktuelle
              Ereignisse, Wissenschaftsschranke (Art. 24d URG) und
              vorübergehende Vervielfältigungen (Art. 24a URG).
    \end{itemize}
\end{definition}

\begin{remark}
    \textbf{Digital Rights Management (DRM, Art. 39a \& 39c URG):}
    \begin{itemize}
        \item Das Umgehen wirksamer technischer Schutzmassnahmen ist verboten.
        \item Elektronische Informationen zur Rechteverwaltung (z.B. digitale
              Wasserzeichen, Copyright- Vermerke) dürfen weder entfernt noch
              verändert werden.
    \end{itemize}
\end{remark}

\subsection{KI und Urheberrecht}

\begin{remark}
    \textbf{Training von KI-Sprachmodellen (LLMs):}
    \begin{itemize}
        \item Das Training dient in der Regel nicht dem "menschlichen Genuss"
              des Werks und führt bei hinreichender Zerlegung/Transformation
              nicht zu einer klassischen Kopie im Modell (Werk verblasst).
        \item Mögliche Schranken: Betrieblicher Eigengebrauch (Art. 19 URG) oder
              Wissenschaftsschranke (Art. 24d URG), sofern kein Re-Memorisieren
              stattfindet.
        \item \textbf{KI-Outputs:} KI-generierter Content ist nur
              urheberrechtlich geschützt, wenn der menschliche Prompt dem Output
              eine ausreichend individuelle Prägung verleiht.
    \end{itemize}
\end{remark}

\subsection{Rechtsfolgen und Schutz ausserhalb des URG}

\begin{remark}
    \textbf{Rechtsfolgen bei Urheberrechtsverletzungen:}
    \begin{itemize}
        \item \textbf{Zivilrecht:} Schadenersatz (nach nachgewiesenem Schaden
              oder Lizenzanalogie) sowie Gewinnherausgabe.
        \item \textbf{Strafrecht (Art. 67 ff. URG):} Vorsätzliche Verletzungen
              werden mit Geld- oder Freiheitsstrafe geahndet (inkl.
              Eventualvorsatz und Gehilfenschaft).
    \end{itemize}
\end{remark}

\begin{remark}
    \textbf{Schutz ausserhalb des Urheberrechts:}
    \begin{itemize}
        \item \textbf{Lauterkeitsrecht (Art. 5 lit. c UWG):} Unlauter handelt,
              wer das marktreife Arbeitsergebnis eines anderen ohne angemessenen
              eigenen Aufwand durch technische Reproduktionsverfahren (z.B.
              Screen Scraping von Inseraten) übernimmt und verwertet.
        \item \textbf{Nutzungsbedingungen (AGB):} Binden Besucher von Webseiten
              grundsätzlich nur, wenn eine gültige Vertragsbeziehung zustande
              gekommen ist (z.B. durch Registrierung).
    \end{itemize}
\end{remark}
