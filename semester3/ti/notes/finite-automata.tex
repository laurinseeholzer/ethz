\section{Finite Automata}

\subsection{Symbol, Alphabets and Words}

\begin{definition}
    \textbf{Alphabet:} finite, non-empty set.
\end{definition}

\begin{example}
    \textbf{Examples of Alphabets:}
    \begin{itemize}
        \item Boolean alphabet ${0,1}$
        \item latin alphabet ${a,b,\dots,z}$
        \item logical alphabet ${0,a,x,(,),\dots}$
    \end{itemize}
\end{example}

\begin{definition}
    \textbf{Word:} A word over an alphabet $A$ is a finite sequence of symbols of $A$. 
\end{definition}

\begin{definition}
    The \textbf{lenght} of a word it the number of symbols it uses. There is always an empty word of length zero.
\end{definition}

\begin{remark}
    Words can be concatenated or rewersed to form new words.
    $$kon(x,y) = xy$$
    $$w = a_1 a_2 a_3 \implies w^R = a_3 a_2 a_1$$ 
\end{remark}

\begin{definition}
    The $i$-th iteration of a word $x = ab$ is $x^i = ababab \dots ab$ the word concatenated with itself i times.
\end{definition}

\begin{definition}
    A word $v$ is a partial-, prefix- or suffix-word of a word $w$ if $v$ is contained in $w$, $w$ begins with $v$ or ends with $v$. 
\end{definition}

\begin{remark}
    For a word $x$ and symbol $a$ the expression $|x|_a$ describes the numnber of apperances of $a$ in $x$. 
\end{remark}

\begin{theorem}
    For an alphabet with an order imposed on its symbols ($a_1 \le a_2 \le ...$) we can define the harmonic order of its words as a lexicographic order where
    $$ u  \le v \iff (|u| \le |v|) \lor (|u| = |v| \land u = xs_iu',v = xs_jv'. s_i\le s_j)$$ 
\end{theorem}

\subsection{Formal Languages}

\begin{definition}
    A subset $L \subset \Sigma^*$ is called Language over the alphabet $\Sigma$
\end{definition}

\begin{definition}
    The \textbf{complement} $L^C$ of a Language $L$ is definied as $\Sigma^* \setminus L$
\end{definition}

\begin{definition}
    The \textbf{concatenation} of two languages $L_1, L_2$ is all the set of all $w_1w_2$ where $w_1 \in L_1, w_2 \in L_2$
\end{definition}

\begin{remark}
    We can concatenate a Language with itself (possibly multiple times or zero)
    $$L^0 = \{\lambda\}, L^1 = L, L^2 = LL \dots L^* = \bigcup_{i=0}^\infty L^i, L^+ = LL^*$$
\end{remark}

\begin{lemma}
    Let $L_1,L_2, L_3$ be languages over an alphabet $\Sigma$ then
    $$L_1L_2 \cup L_1L_3 = L_1(L_2 \cup L_3)$$
    $$L_1(L_2 \cap L_3) \subseteq L_1L_2 \cap L_1L_3$$
\end{lemma}

\subsection{Algorithmic Problems}

\begin{remark}
    An algorithm describes a function $\Sigma_{in}^* \to \Sigma_{out}^*$
\end{remark}

\begin{definition}
    Let $\Sigma$ be an alphabet and $L \subseteq \Sigma^*$. The decision problem $(\Sigma,L)$ is defined (for arb $x \in Sigma^*$) as
    $$x \in L \quad \text{or} \quad x \notin L$$
\end{definition}

\begin{remark}
    If an algorithm A correctly decides for any $x$ then we say
    $$A \quad \text{detects the Language} \quad L$$
\end{remark}

\subsection{Finite Automata}

\begin{remark}
    A finite automata (finite state machine) is a set of states together with state transitions on inputs with fixed boolean outputs on each state.
\end{remark}

\begin{definition}
    \textbf{Finite Automata:} is a Quadroupel $M = (Q,\Sigma, \delta, q_0, F)$
    \begin{itemize}
        \item $Q$ is a finite set of states
        \item $\Sigma$ is an alphabet (Input-alphabet)
        \item $\delta$ is a function $Q \times \Sigma \to Q$
        \item $q_0 \in Q$ is the starting-state
        \item $F \subseteq Q$ is the set of accepting (true) states
    \end{itemize}
\end{definition}

\begin{definition}
    \textbf{Configuration:} of an Automata is an Elment of $Q \times \Sigma^*$
    \begin{itemize}
        \item \textbf{Starting-Configuraiton:} $C_0 = (q_0, x) | x \in \Sigma^*$
        \item \textbf{End-Configruation:} $C_n = (q_0, \lambda)$ where its accepting if $C_n \in F \times \{\lambda\}$ 
    \end{itemize}
\end{definition}

\begin{remark}
    A calculation can be seen as a finite sequence of configurations $C_1, C_2 \dots$ such that for every $C_i \to_{s_M} C_{i+1}$ with steps $s_M$ in our machine $M$
    $$(q_0, 01001) \vdash_{M} (q_x, 1001) \vdash_{M} (q_y, 001) \dots$$
    where a step is a relation on configurations
    $$\vdash_{M} \subseteq (Q \times \Sigma^*) \times (Q \times \Sigma^*)$$ 
\end{remark}


\begin{definition}
    The \textbf{transitive hull of transitions} describes the calculation over multiple steps:
    $$(q,w) \vdash_{M}^{*} (p,u) \iff (q = p \land w = u) \lor \exists k \in \mathbb{N}^{+} \text{ s.t. }$$
    $$w = a_1 \dots a_k u \quad \text{with} \quad a_i \in \Sigma \quad \text{and}$$
    $$\exists r_1, \dots, r_{k-1} \in Q \text{ s.t. } (q,w) \vdash_{M} (r_1, a_2 \dots a_k u) \vdash_{M} (r_2, a_3 \dots a_k u) \dots \vdash_{M} (p,u)$$
\end{definition}

\begin{definition}
    The \textbf{extended transition function} evaluates the end state of a word over multiple transitions:
    $$\tilde{\delta}: Q \times \Sigma^{*} \to Q$$
    $$\tilde{\delta}(q, \lambda) = q \quad \forall q \in Q$$
    $$\tilde{\delta}(q, wa) = \delta(\tilde{\delta} (q,w), a) \quad \forall a \in \Sigma, w \in \Sigma^{*}, q \in Q$$
\end{definition}

\begin{remark}
    States can be grouped into equivalence classes containing all elements with the same end state. This is useful for proofs regarding the language of finite automata, which are typically done through these classes and induction.
\end{remark}

\begin{definition}
    \textbf{Product Automata:} Used to determine the intersection, union or exclusion ($\odot$) of two finite automata (their languages), built with the state space $Q = Q_1 \times Q_2$.
    $$M_1 = (Q_1, \Sigma, \delta_1, q_{01}, F_1), \quad M_2 = (Q_2, \Sigma, \delta_2, q_{02}, F_2)$$
    $$\implies M = (Q_1 \times Q_2, \delta((x,y), a) = (\delta_1(x, a), \delta_2(y, a)), (q_01, q_02), F_1 \odot F_2)$$
\end{definition}

\begin{remark}
    The correctness of this product automata can be proven by induction over the word length.
\end{remark}

\subsection{Non-regularity}

\begin{definition}
    A language $L$ is called \textbf{regular} if there exists a finite automata that accepts/detects the language $L$ and \textbf{non-reular} else.
\end{definition}


\begin{lemma}
    Let $A = (Q, \Sigma, \delta, q_0, F)$ be a finite automata and let $x,y \in \Sigma^*$ such that
    $$\tilde{\delta}(q_0,x) = \tilde{\delta}(q_0, y)$$
    then there exists $z \in \Sigma^*$ such that
    $$Kl[xz] = Kl[yz]$$
\end{lemma}

\begin{remark}
    This illustrates that finite machines are finite hence can not distinguish beetween $xz$ and $yz$ once they are in a certain state $q_r$, they do not know how they got there.
\end{remark}

\begin{proof}
    \textbf{Proof of Non-Regularity:} Let $L = \{0^n1^n | n \in \mathbb{N}\}$ and assume $L \in L_{EA}$ ($L$ is regular) hence there exists a finite automata $A$ such that $L(A) = L$.
    Let $m = |Q|$ and observe the words $x \in \{0^n | n \leq m+1 \}$. Because this are more words than $A$ has states there exists $i \neq j \in [0, m+1] \cap \mathbb{N}$ such that
    $$\tilde{\delta}(q_0, 0^i) = \tilde{\delta}(q_0, 0^j)$$
    Hence by the above lemma the words $0^i1^i, 0^j1^i$ will both be accepted which contradicts the assumption that $A$ detects $L$ because $0^j1^i \notin L$.
\end{proof}
