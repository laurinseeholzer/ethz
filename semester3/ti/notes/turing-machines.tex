\section{Turingmaschinen und Grundlagen der Berechenbarkeit}

\subsection{Das Standardmodell (Deterministische 1-Band-Turingmaschine)}

Die Turingmaschine stellt das fundamentale mathematische Modell eines universellen Computers dar. Im Gegensatz zu endlichen Automaten verfügt sie über einen unbeschränkten, beidseitig adressierbaren (im Standardmodell einseitig unendlichen) Arbeitsspeicher in Form eines Bandes sowie einen beweglichen Lese-/Schreibkopf.

\begin{definition}[Deterministische 1-Band-Turingmaschine]
    Eine deterministische 1-Band-Turingmaschine (DTM) ist ein 7-Tupel
    \[
        M = (Q, \Sigma, \Gamma, \delta, q_0, q_{\mathrm{accept}}, q_{\mathrm{reject}}),
    \]
    wobei:
    \begin{itemize}
        \item $Q$ eine endliche, nichtleere Menge von \textbf{Zuständen} ist,
        \item $\Sigma$ das endliche \textbf{Eingabealphabet} ist,
        \item $\Gamma$ das endliche \textbf{Bandalphabet} ist mit $\Sigma \subsetneq \Gamma$,
        \item $\text{\textcent} \in \Gamma \setminus \Sigma$ das linke \textbf{Randbegrenzungssymbol} (Left-End-Marker) ist,
        \item $\sqcup \in \Gamma \setminus (\Sigma \cup \{\text{\textcent}\})$ das \textbf{Blanksymbol} (Leersymbol) ist,
        \item $q_0 \in Q$ der \textbf{Startzustand} ist,
        \item $q_{\mathrm{accept}} \in Q$ der \textbf{akzeptierende Stoppzustand} ist,
        \item $q_{\mathrm{reject}} \in Q$ der \textbf{verwerfende Stoppzustand} ist ($q_{\mathrm{accept}} \neq q_{\mathrm{reject}}$),
        \item $\delta$ die (partielle) \textbf{Übergangsfunktion} ist:
        \[
            \delta : (Q \setminus \{q_{\mathrm{accept}}, q_{\mathrm{reject}}\}) \times \Gamma \to Q \times \Gamma \times \{\mathrm{L}, \mathrm{R}\}.
        \]
    \end{itemize}
    \textbf{Randbegrenzungskonvention:} Das Symbol $\text{\textcent}$ darf weder überschrieben noch nach links überschritten werden. Für alle $q \in Q \setminus \{q_{\mathrm{accept}}, q_{\mathrm{reject}}\}$ gilt daher stets:
    \[
        \delta(q, \text{\textcent}) = (p, \text{\textcent}, \mathrm{R}) \quad \text{für ein } p \in Q.
    \]
\end{definition}

\begin{definition}[Konfiguration und Konfigurationsübergang]
    Eine \textbf{Konfiguration} beschreibt den vollständigen Gesamtzustand der Turingmaschine zu einem diskreten Zeitschritt. Sie umfasst den aktuellen Zustand, den gesamten nicht-blanken Bandinhalt sowie die Kopfposition.
    
    Formell wird eine Konfiguration als Wort aus $\text{\textcent}\Gamma^* Q \Gamma^+ \cup Q\text{\textcent}\Gamma^+$ dargestellt, wobei der Zustand $q \in Q$ unmittelbar links von demjenigen Symbol steht, das der Kopf aktuell liest:
    \begin{itemize}
        \item $C = q \text{\textcent} u$: Der Kopf liest die linke Randbegrenzung $\text{\textcent}$.
        \item $C = \text{\textcent} u q a v$ mit $u, v \in \Gamma^*, a \in \Gamma$: Der Kopf liest das Symbol $a$.
    \end{itemize}
    Die Relation des \textbf{Ein-Schritt-Übergangs} $\vdash_M$ ist für zwei Konfigurationen $C_1, C_2$ definiert durch:
    \begin{itemize}
        \item Rechtsbewegung: $\text{\textcent} u q a v \vdash_M \text{\textcent} u b p v$ falls $\delta(q, a) = (p, b, \mathrm{R})$ (für $v = \lambda$ wird implizit ein $\sqcup$ gelesen: $\text{\textcent} u q a \vdash_M \text{\textcent} u b p \sqcup$).
        \item Linksbewegung: $\text{\textcent} u c q a v \vdash_M \text{\textcent} u p c b v$ falls $\delta(q, a) = (p, b, \mathrm{L})$.
    \end{itemize}
    Die reflexive und transitive Hülle von $\vdash_M$ wird mit $\vdash_M^*$ notiert.
    
    Für eine Eingabe $w \in \Sigma^*$ ist die \textbf{Startkonfiguration} definiert als:
    \[
        C_0(w) = q_0 \text{\textcent} w \quad (\text{bzw. } q_0 \text{\textcent} \sqcup \text{ falls } w = \lambda).
    \]
    Eine Konfiguration heißt \textbf{Stoppkonfiguration}, falls ihr Zustand in $\{q_{\mathrm{accept}}, q_{\mathrm{reject}}\}$ liegt.
\end{definition}

\begin{definition}[Sprachakzeptanz und Berechenbarkeitsklassen]
    Eine Turingmaschine $M$ \textbf{akzeptiert} ein Wort $w \in \Sigma^*$, falls
    \[
        C_0(w) \vdash_M^* C_{\mathrm{stop}} \quad \text{mit Zustand } q_{\mathrm{accept}}.
    \]
    $M$ \textbf{verwirft} $w$, falls $C_0(w) \vdash_M^* C_{\mathrm{stop}}$ mit Zustand $q_{\mathrm{reject}}$. 
    
    Die von $M$ akzeptierte Sprache ist
    \[
        L(M) = \{w \in \Sigma^* \mid M \text{ akzeptiert } w\}.
    \]
    Wir unterscheiden zwei fundamentale Sprachklassen:
    \begin{enumerate}
        \item \textbf{Rekursiv aufzählbare / semi-entscheidbare Sprachen ($\mathcal{L}_{\mathrm{RE}}$):}
        Eine Sprache $L \subseteq \Sigma^*$ liegt in $\mathcal{L}_{\mathrm{RE}}$, falls eine TM $M$ existiert mit $L = L(M)$. Für $w \notin L$ darf $M$ in $q_{\mathrm{reject}}$ stoppen oder in eine Endlosschleife geraten ($M$ divergiert).
        \item \textbf{Rekursive / entscheidbare Sprachen ($\mathcal{L}_{\mathrm{rec}}$):}
        Eine Sprache $L \subseteq \Sigma^*$ liegt in $\mathcal{L}_{\mathrm{rec}}$, falls eine TM $M$ existiert mit $L = L(M)$, die auf \textbf{jeder} Eingabe $w \in \Sigma^*$ garantiert nach endlich vielen Schritten in $q_{\mathrm{accept}}$ oder $q_{\mathrm{reject}}$ stoppt ($M$ ist ein \textbf{Entscheider}).
    \end{enumerate}
    Offensichtlich gilt $\mathcal{L}_{\mathrm{rec}} \subseteq \mathcal{L}_{\mathrm{RE}}$.
\end{definition}

\begin{definition}[Berechenbare Funktionen]
    Eine Funktion $f : \Sigma^* \to \Gamma^*$ heißt:
    \begin{itemize}
        \item \textbf{partiell berechenbar}, falls eine DTM $M$ existiert, die auf Eingabe $w \in \mathrm{Def}(f)$ in $q_{\mathrm{accept}}$ mit Bandinhalt $\text{\textcent} f(w) \sqcup \sqcup \dots$ stoppt, und für $w \notin \mathrm{Def}(f)$ entweder nicht stoppt oder in $q_{\mathrm{reject}}$ stoppt.
        \item \textbf{(total) berechenbar}, falls $f$ partiell berechenbar ist und $\mathrm{Def}(f) = \Sigma^*$.
    \end{itemize}
\end{definition}

\begin{beispiel}[Palindromerkennung auf einer 1-Band-TM]
    Betrachte $L = \{w \in \{0, 1\}^* \mid w = w^R\}$. Ein 1-Band-Algorithmus liest das erste Symbol hinter $\text{\textcent}$, überschreibt es mit $\sqcup$, merkt sich das Bit im Zustand, läuft nach rechts bis zum Wortende, vergleicht das letzte Symbol, löscht es und läuft zurück zu $\text{\textcent}$. Dies wiederholt sich, bis das Wort leer ist. Die Laufzeit auf Eingabe der Länge $n$ ist $\Theta(n^2)$.
\end{beispiel}

\subsection{Erweiterungen und Äquivalenzsätze}

In der Praxis modellieren wir Algorithmen gerne mit mehreren Bändern oder nichtdeterministischen Verzweigungen. Wir beweisen nun, dass diese Modellerweiterungen die Klasse der berechenbaren Probleme nicht erweitern.

\begin{definition}[Mehrband-Turingmaschine (MTM)]
    Eine deterministische $k$-Band-Turingmaschine besitzt:
    \begin{itemize}
        \item Ein separates, unidirektionales oder bidirektionales \textbf{Nur-Lese-Eingabeband} mit Randmarkierungen $\text{\textcent} w \$$,
        \item $k \ge 1$ separate, beidseitig (oder einseitig) unendliche \textbf{Arbeitsbänder} mit jeweils eigenem Lese-/Schreibkopf.
    \end{itemize}
    Die Übergangsfunktion liest in jedem Schritt die Symbole aller $k+1$ Köpfe und bestimmt für jedes Band unabhängig ein neues Symbol sowie eine Kopfbewegung $\{\mathrm{L}, \mathrm{R}, \mathrm{N}\}$.
\end{definition}

\begin{theorem}[Äquivalenz von Mehrband- und 1-Band-TMs]
    Zu jeder $k$-Band-Turingmaschine $M$, die auf Eingaben der Länge $n$ in Zeit $T(n)$ anhält, existiert eine äquivalente 1-Band-Turingmaschine $M'$ mit $L(M') = L(M)$, die in Zeit $O((T(n))^2)$ arbeitet.
\end{theorem}

\begin{proof}
    \textbf{Beweisidee:} Wir simulieren die $k$ Arbeitsbänder und das Eingabeband durch eine \emph{Spurtechnik} (Tracks) auf einem einzigen Band, wobei virtuelle Kopfpositionen durch markierte Symbole repräsentiert werden.
    
    Das Bandalphabet $\Gamma'$ von $M'$ besteht aus Tupeln: Für jedes der $k+1$ Bänder führt $\Gamma'$ zwei Komponenten:
    \begin{enumerate}
        \item Das auf dem Band gespeicherte Symbol $a \in \Gamma$.
        \item Ein Markierungsbit $\hat{a} \in \{0, 1\}$, das anzeigt, ob sich der virtuelle Kopf des jeweiligen Bandes auf dieser Zelle befindet.
    \end{enumerate}
    Ein Rechenschritt von $M$ wird von $M'$ wie folgt simuliert:
    \begin{enumerate}
        \item $M'$ scannt das Band von ganz links nach rechts bis zur am weitesten rechts stehenden Kopfmarkierung und sammelt im Zustand die $k+1$ gelesenen Symbole ein.
        \item Entsprechend der Übergangsfunktion $\delta$ von $M$ läuft $M'$ zurück nach links, aktualisiert die Bandinhalte der Spuren und verschiebt die Kopfmarkierungen um höchstens eine Zelle nach links, rechts oder lässt sie neutral.
        \item Erfordert ein Schritt das Erweitern des Bandes nach rechts, werden die Spuren um eine Zelle verschoben.
    \end{enumerate}
    Nach $t$ Schritten können die Köpfe von $M$ höchstens $t$ Zellen vom linken Rand entfernt sein. Ein Sweep von $M'$ über alle markierten Zellen benötigt daher höchstens $O(t)$ Schritte. Für $T(n)$ Schritte von $M$ ergibt sich die Gesamtlaufzeit:
    \[
        \sum_{t=1}^{T(n)} O(t) = O((T(n))^2).
    \]
\end{proof}

\begin{definition}[Nichtdeterministische Turingmaschine (NTM)]
    Eine nichtdeterministische Turingmaschine ist ein 7-Tupel $N = (Q, \Sigma, \Gamma, \delta, q_0, q_{\mathrm{accept}}, q_{\mathrm{reject}})$, bei dem die Übergangsfunktion durch eine Relation ersetzt ist:
    \[
        \delta : (Q \setminus \{q_{\mathrm{accept}}, q_{\mathrm{reject}}\}) \times \Gamma \to \mathcal{P}(Q \times \Gamma \times \{\mathrm{L}, \mathrm{R}, \mathrm{N}\}).
    \]
    Ein Wort $w \in \Sigma^*$ wird von $N$ \textbf{akzeptiert}, wenn im Berechnungsbaum mindestens ein endlicher Berechnungspfad existiert, der von $C_0(w)$ in eine akzeptierende Stoppkonfiguration führt:
    \[
        L(N) = \{w \in \Sigma^* \mid \exists \text{ endliche Folge } C_0(w) \vdash_N C_1 \vdash_N \dots \vdash_N C_m \text{ mit Zustand } q_{\mathrm{accept}}\}.
    \]
\end{definition}

\begin{theorem}[Simulation von NTMs durch DTMs]
    Jede nichtdeterministische Turingmaschine $N$ kann durch eine deterministische 3-Band-Turingmaschine $D$ simuliert werden, d.\,h.\ $L(D) = L(N)$. Insbesondere gilt:
    \[
        \mathcal{L}(NTM) = \mathcal{L}(DTM) = \mathcal{L}_{\mathrm{RE}}.
    \]
\end{theorem}

\begin{proof}
    \textbf{Beweisidee:} Der Berechnungsbaum von $N$ wird systematisch mittels \emph{Breitensuche} traversiert, um zu verhindern, dass die deterministische Maschine in einem unendlich langen, nicht-akzeptierenden Pfad verhungert.
    
    Sei $b = \max_{(q, a)} |\delta(q, a)|$ der maximale Verzweigungsgrad des Berechnungsbaumes. Jeder Rechenpfad der Länge $m$ lässt sich eindeutig durch eine Ziffernfolge $d_1 d_2 \dots d_m \in \{1, \dots, b\}^m$ adressieren.
    
    $D$ verwendet drei Bänder:
    \begin{itemize}
        \item \textbf{Band 1:} Speichert permanent die unveränderte Eingabe $w$.
        \item \textbf{Band 2 (Adressband):} Erzeugt sukzessive alle Wörter aus $\{1, \dots, b\}^*$ in quasielexikographischer Reihenfolge:
        \[
            \lambda, 1, 2, \dots, b, 11, 12, \dots, 1b, \dots, bb, 111, \dots
        \]
        \item \textbf{Band 3 (Simulationsband):} Kopiert $w$ von Band 1 und führt die Berechnung von $N$ deterministisch entlang der auf Band 2 vorgegebenen Verzweigungen aus.
    \end{itemize}
    Erreicht die Simulation auf Band 3 den Zustand $q_{\mathrm{accept}}$, hält $D$ an und akzeptiert. Ist ein Pfad ungültig oder endet in $q_{\mathrm{reject}}$ vor Abarbeitung aller Adressziffern, bricht $D$ die Simulation ab, löscht Band 3, inkrementiert Band 2 zur nächsten Pfadadresse und wiederholt den Zyklus.
\end{proof}

\begin{important}[Warum Breitensuche und keine Tiefensuche?]
    Für entscheidbare Sprachen mit garantierter Zeitschranke wäre eine Tiefensuche denkbar. Für allgemeine semi-entscheidbare Sprachen ($\mathcal{L}_{\mathrm{RE}}$) versagt die Tiefensuche (DFS) jedoch fundamental: Besitzt der Berechnungsbaum einen linken Ast unendlicher Länge ohne akzeptierende Konfiguration, würde sich eine DFS-Simulation für immer in diesem unendlichen Pfad verlieren und einen existierenden akzeptierenden Zustand in einem Nachbarast niemals entdecken! Die Breitensuche garantiert, dass jeder endliche Rechenpfad nach endlich vielen Schritten vollständig exploriert wird.
\end{important}

\begin{theorem}[Church-Turing-These]
    Die Klasse der intuitiv im algorithmischen Sinne berechenbaren Funktionen stimmt exakt mit der Klasse der Turing-berechenbaren Funktionen überein.
\end{theorem}

\begin{remark}[Wissenschaftstheoretischer Status]
    Die Church-Turing-These ist kein mathematischer Satz im formalen Sinn, sondern eine empirisch fundierte Hypothese über die physikalische Realisierbarkeit von Informationsverarbeitung. Sämtliche historisch vorgeschlagenen Formalisierungen (Gödels $\mu$-rekursive Funktionen, Churchs $\lambda$-Kalkül, Postsche Produktionssysteme, Registermaschinen, Random-Access-Machines, moderne Programmiersprachen) erwiesen sich als beweisbar äquivalent zur Turingmaschine.
\end{remark}

\subsection{Kodierung von Turingmaschinen und die Universelle Turingmaschine}

Um Turingmaschinen selbst als Eingaben für Algorithmen verarbeiten zu können, müssen wir sie in eine standardisierte Bitfolge übersetzen (Gödelisierung).

\begin{definition}[Standardkodierung einer Turingmaschine]
    Sei $M = (Q, \Sigma, \Gamma, \delta, q_0, q_{\mathrm{accept}}, q_{\mathrm{reject}})$. Ohne Einschränkung nummerieren wir:
    \begin{itemize}
        \item Zustände: $Q = \{q_1, q_2, \dots, q_{|Q|}\}$ mit $q_0 = q_1$, $q_{\mathrm{accept}} = q_2$, $q_{\mathrm{reject}} = q_3$.
        \item Symbole: $\Gamma = \{s_1, s_2, \dots, s_{|\Gamma|}\}$ mit $s_1 = 0$, $s_2 = 1$, $s_3 = \sqcup$, $s_4 = \text{\textcent}$.
        \item Richtungen: $D_1 = \mathrm{L}$, $D_2 = \mathrm{R}$.
    \end{itemize}
    Jede Übergangsregel $\delta(q_i, s_j) = (q_k, s_l, D_m)$ wird als Zeichenkette über $\{0, 1, \#\}$ kodiert:
    \[
        \mathrm{kod}(\delta_r) = 0^i \# 0^j \# 0^k \# 0^l \# 0^m.
    \]
    Die gesamte Maschine $M$ wird durch Aneinanderreihung aller $R$ Übergänge kodiert:
    \[
        \mathrm{Kod}'(M) = \#\# \mathrm{kod}(\delta_1) \#\# \mathrm{kod}(\delta_2) \#\# \dots \#\# \mathrm{kod}(\delta_R) \#\#.
    \]
    Durch den injektiven Homomorphismus $h : \{0, 1, \#\}^* \to \{0, 1\}^*$ mit
    \[
        h(0) = 00, \quad h(1) = 11, \quad h(\#) = 01
    \]
    erhalten wir eine reine Binärkodierung $\mathrm{Kod}(M) = h(\mathrm{Kod}'(M)) \in \{0, 1\}^*$.
\end{definition}

\begin{lemma}[Kanonische Aufzählung aller Turingmaschinen]
    Die Menge aller syntaktisch gültigen Turingmaschinen-Kodierungen ist eine entscheidbare Teilmenge $\mathrm{KodTM} \subset \{0, 1\}^*$. Indem man jedem ungültigen Binärwort per Konvention eine feste, sofort verwerfende Turingmaschine $M_\emptyset$ zuordnet, induziert die quasielexikographische Ordnung auf $\{0, 1\}^*$ eine bijektive Abzählung aller Turingmaschinen:
    \[
        M_1, M_2, M_3, M_4, \dots
    \]
\end{lemma}

\begin{theorem}[Die Universelle Turingmaschine]
    Es existiert eine universelle deterministische Turingmaschine $U$, die auf Eingabe von $\mathrm{Kod}(M) \# w$ das Verhalten der Turingmaschine $M$ auf $w$ exakt simuliert:
    \[
        L(U) = L_U = \{\mathrm{Kod}(M) \# w \mid M \text{ akzeptiert } w\} \in \mathcal{L}_{\mathrm{RE}}.
    \]
\end{theorem}

\begin{proof}
    \textbf{Beweisidee:} $U$ verwaltet den Bandinhalt von $M$ und den Zustand von $M$ auf getrennten Arbeitsspuren und sucht für jeden Einzelschritt von $M$ die passende Übergangsregel in der Kodierung.
    
    $U$ arbeitet als 3-Band-TM:
    \begin{itemize}
        \item \textbf{Band 1:} Enthält die Eingabe $\mathrm{Kod}(M) \# w$. Zunächst wird die syntaktische Korrektheit von $\mathrm{Kod}(M)$ verifiziert.
        \item \textbf{Band 2:} Simuliert das Arbeitsband von $M$. Zu Beginn wird $w$ auf Band 2 kopiert.
        \item \textbf{Band 3:} Speichert den aktuellen Zustand $q_i$ von $M$ in Unärkodierung $0^i$ (initialisiert mit $0^1$ für $q_0$).
    \end{itemize}
    In jeder Simulationsrunde liest $U$ das Symbol $s_j$ unter dem Kopf von Band 2, durchsucht Band 1 nach dem Teilblock $\#\# 0^i \# 0^j \# 0^k \# 0^l \# 0^m \#\#$, ersetzt auf Band 3 den Zustand durch $0^k$, überschreibt auf Band 2 das Symbol $s_j$ mit $s_l$ und bewegt den Kopf von Band 2 gemäß $D_m$.
    Erreicht Band 3 den Zustand $0^2$ ($q_{\mathrm{accept}}$), hält $U$ und akzeptiert. Erreicht Band 3 $0^3$ ($q_{\mathrm{reject}}$) oder existiert kein passender Übergang, hält $U$ und verwirft.
\end{proof}
