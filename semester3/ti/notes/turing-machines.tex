\section{Turing Machines}

\subsection{Basic Concept}

\begin{remark}
    A Turing Machine (TM) is a fundamental generalization of finite automata:
    \begin{itemize}
        \item \textbf{Finite Control:} Like a finite automaton, a TM possesses a finite set of internal states $Q$.
        \item \textbf{Unbounded Memory (Tape):} Instead of a one-way read-only input, a TM has an infinite (or semi-infinite) tape divided into discrete cells, serving as both input medium and unbounded memory.
        \item \textbf{Read/Write Head:} A movable head that can read the symbol in the current cell, overwrite it with a new symbol, and move either left ($\mathrm{L}$) or right ($\mathrm{R}$) along the tape.
    \end{itemize}
\end{remark}

\subsection{Formal Definition}

\begin{definition}
    \textbf{Turing Machine:} A deterministic Turing Machine is a 7-tuple
    $$
        M = (Q, \Sigma, \Gamma, \delta, q_0, q_{accept}, q_{reject})
    $$
    where:
    \begin{itemize}
        \item $Q$ is a finite, non-empty set of states.
        \item $\Sigma$ is a finite input alphabet.
        \item $\Gamma$ is a finite tape alphabet such that $\Sigma \subset \Gamma$. It contains:
              \begin{itemize}
                  \item the \textbf{blank symbol} $\sqcup \in \Gamma \setminus \Sigma$ (representing empty cells)
                  \item the \textbf{left-end marker} $\text{\textcent} \in \Gamma \setminus \Sigma$ (marking the left boundary of the tape)
              \end{itemize}
        \item $\delta$ is the transition function:
              $$
                  \delta : (Q \setminus \{q_{accept}, q_{reject}\}) \times \Gamma \to Q \times \Gamma \times \{\mathrm{L}, \mathrm{R}\}
              $$
              with the convention that upon scanning the left-end marker $\text{\textcent}$, the machine cannot overwrite it and must move right:
              $$
                  \delta(q, \text{\textcent}) = (p, \text{\textcent}, \mathrm{R}) \quad \text{for some } p \in Q
              $$
        \item $q_0 \in Q$ is the starting state.
        \item $q_{accept} \in Q$ is the accepting halt state.
        \item $q_{reject} \in Q$ is the rejecting halt state ($q_{accept} \neq q_{reject}$).
    \end{itemize}
\end{definition}

\subsection{Configurations}

\begin{definition}
    \textbf{Configuration:} A configuration of a Turing Machine captures its current global state, consisting of the tape contents up to the last relevant cell, the current state $q$, and the head position.
    Formally, the set of all configurations is:
    $$
        \mathrm{Konf}(M) = \{\text{\textcent}\} \cdot \Gamma^* \cdot Q \cdot \Gamma^+ \;\cup\; Q \cdot \{\text{\textcent}\} \cdot \Gamma^+
    $$
    In this word representation, the state symbol $q \in Q$ is written immediately to the left of the tape cell currently scanned by the read/write head:
    \begin{itemize}
        \item If the head scans the left-end marker $\text{\textcent}$:
              $$
                  C = q \text{\textcent} u \in Q \cdot \{\text{\textcent}\} \cdot \Gamma^+
              $$
        \item If the head scans cell $i \ge 1$ with tape prefix $\text{\textcent} u$:
              $$
                  C = \text{\textcent} u q v \in \{\text{\textcent}\} \cdot \Gamma^* \cdot Q \cdot \Gamma^+
              $$
    \end{itemize}
\end{definition}

\begin{definition}
    \textbf{Start Configuration:} For an input word $x \in \Sigma^*$, the start configuration is:
    $$
        C_0(x) = \begin{cases}
            q_0 \text{\textcent} x \sqcup & \text{if } x \neq \lambda \\
            q_0 \text{\textcent} \sqcup & \text{if } x = \lambda
        \end{cases}
    $$
    The machine starts in state $q_0$ scanning the left-end marker $\text{\textcent}$, followed by the input $x$ and blank symbols $\sqcup$.
\end{definition}

\subsection{Computation and Steps}

\begin{definition}
    \textbf{Computation Step:} A single step is a transition relation $\vdash_M$ between configurations:
    \begin{itemize}
        \item \textbf{Move Right ($\mathrm{R}$):} If $\delta(q, a) = (p, b, \mathrm{R})$, then for all $u \in \Gamma^*, v \in \Gamma^+$:
              $$
                  u q a v \vdash_M u b p v
              $$
              (if $v = \lambda$, the cell is implicitly filled with $\sqcup$).
        \item \textbf{Move Left ($\mathrm{L}$):} If $\delta(q, a) = (p, b, \mathrm{L})$, then for all $u \in \Gamma^*, c \in \Gamma, v \in \Gamma^+$:
              $$
                  u c q a v \vdash_M u p c b v
              $$
    \end{itemize}
    We denote by $\vdash_M^*$ the reflexive and transitive closure of $\vdash_M$ (computation over multiple steps).
\end{definition}

\begin{definition}
    \textbf{Computation (Calculation):} A computation of $M$ on input $x$ is a sequence of configurations $C_0, C_1, C_2, \dots$ where $C_0 = C_0(x)$ and $C_i \vdash_M C_{i+1}$ for all $i \ge 0$.
    \begin{itemize}
        \item \textbf{Halting:} The computation terminates if the machine reaches a halting state ($q_{accept}$ or $q_{reject}$), since $\delta$ is not defined for these states.
        \item \textbf{Infinite Loop:} If no halting state is ever reached, the computation runs forever ($M$ does not terminate on $x$).
    \end{itemize}
\end{definition}

\subsection{Multi-Tape Turing Machines}

\begin{definition}
    \textbf{Multi-Tape Turing Machine ($k$-Tape TM):} A $k$-tape Turing Machine ($k \ge 1$) consists of a finite control and $k$ independent, semi-infinite tapes, each equipped with its own independently moving read/write head:
    $$
        M = (Q, \Sigma, \Gamma, \delta, q_0, q_{accept}, q_{reject})
    $$
    where the transition function is defined as:
    $$
        \delta : (Q \setminus \{q_{accept}, q_{reject}\}) \times \Gamma^k \to Q \times \Gamma^k \times \{\mathrm{L}, \mathrm{R}, \mathrm{S}\}^k
    $$
    In a single computation step, the machine reads the $k$-tuple of symbols currently under the $k$ heads $(a_1, \dots, a_k) \in \Gamma^k$, transitions to a new state $p \in Q$, writes new symbols $(b_1, \dots, b_k) \in \Gamma^k$, and moves each head independently according to $(D_1, \dots, D_k) \in \{\mathrm{L}, \mathrm{R}, \mathrm{S}\}^k$ (where $\mathrm{S}$ denotes staying stationary).
\end{definition}

\begin{remark}
    \textbf{Start Configuration:} On input $x \in \Sigma^*$, Tape 1 contains the input word preceded by the left-end marker $\text{\textcent} x \sqcup \dots$, while all other $k-1$ work tapes are initially empty ($\text{\textcent} \sqcup \dots$). All $k$ heads initially scan the left-end marker $\text{\textcent}$.
\end{remark}

\begin{remark}
    \textbf{Why Use Multi-Tape Machines?}
    \begin{itemize}
        \item \textbf{Natural Algorithm Design:} A multi-tape TM provides a much more intuitive programming paradigm than a single-tape TM. Dedicated tapes can be designated for specific roles (e.g., an input tape, one or more scratch/work tapes, and an output tape).
        \item \textbf{Modularity:} Multiple tapes allow implementing modular subroutines (e.g., copying strings, arithmetic, or counting) without continually traversing and corrupting the primary input.
    \end{itemize}
\end{remark}

\begin{theorem}
    \textbf{Equivalence of Single-Tape and Multi-Tape TMs:} Multi-tape Turing machines have exactly the same computational power as standard single-tape Turing machines. A language $L$ is Turing-recognizable (resp. decidable) by a $k$-tape TM if and only if it is Turing-recognizable (resp. decidable) by a single-tape TM:
    $$
        \mathcal{L}(\text{Multi-Tape TM}) = \mathcal{L}(\text{Single-Tape TM})
    $$
\end{theorem}

\begin{remark}
    \textbf{Simulation via Tracks (Spurtechnik):} A single-tape TM $S$ can simulate a $k$-tape TM $M$ by dividing its single tape into $2k$ tracks using an expanded tape alphabet $\Gamma_S \cong (\Gamma \times \{0, 1\})^k$:
    \begin{itemize}
        \item For each tape $i \in \{1, \dots, k\}$, one track stores the contents of tape $i$, and a second track stores a head marker flag (a bit $1$ or a marked symbol $\hat{a}$) indicating the current position of head $i$.
        \item To simulate one step of $M$, machine $S$ sweeps across the tape from the leftmost head marker to the rightmost head marker, records all $k$ scanned symbols in its finite control, applies $\delta$, and sweeps back to write the replacement symbols and shift the head markers.
    \end{itemize}
\end{remark}

\begin{definition}
    \textbf{Church-Turing Thesis (Church'sche These):}
    \begin{quote}
    \emph{Every algorithmically computable function or intuitive effective procedure can be carried out by a standard Turing Machine.}
    \end{quote}
\end{definition}

\begin{remark}
    \textbf{Significance and Robustness:}
    \begin{itemize}
        \item Augmenting a Turing machine with multiple tapes, multi-dimensional tapes, random-access memory (RAMs), or non-determinism does not expand the class of decidable languages. This demonstrates the exceptional robustness of the Turing machine model.
        \item Independent models of computability ($\lambda$-calculus, $\mu$-recursive functions, RAMs, and modern programming languages) define the exact same class of computable functions.
        \item \textbf{Extended Church-Turing Thesis:} Any physically realistic computational model can be simulated by a deterministic single-tape Turing machine with at most a polynomial slowdown. Because simulating a $k$-tape TM on a 1-tape TM incurs only a quadratic slowdown, the multi-tape TM fully complies with the Extended Church-Turing Thesis.
    \end{itemize}
\end{remark}

\begin{definition}
    \textbf{Time Complexity:} Let $T : \mathbb{N} \to \mathbb{N}$. A Turing Machine $M$ runs in time $T(n)$ if for every input word $x \in \Sigma^*$ of length $n = |x|$, the computation of $M$ on $x$ halts within at most $T(n)$ steps.
\end{definition}

\begin{theorem}
    \textbf{Quadratic Simulation Theorem (Hartmanis \& Stearns):} Let $M$ be a $k$-tape Turing machine that decides a language $L$ in time $T(n)$. Then there exists a single-tape Turing machine $S$ that decides $L$ in time $\mathcal{O}(T(n)^2)$:
    $$
        \mathrm{TIME}_{k\text{-tape}}(T(n)) \subseteq \mathrm{TIME}_{1\text{-tape}}(T(n)^2)
    $$
\end{theorem}

\begin{remark}
    \textbf{Derivation of Quadratic Overhead:}
    \begin{itemize}
        \item In $T(n)$ steps, each head of $M$ can travel at most distance $T(n)$ from the left-end marker. Hence, all $k$ head markers on the single tape of $S$ are contained within a span of at most $k \cdot T(n) = \mathcal{O}(T(n))$ cells.
        \item For every single step of $M$, machine $S$ sweeps over this span to read the symbols under the $k$ head markers and sweeps back to update them, requiring $\mathcal{O}(T(n))$ steps per simulated step.
        \item Total simulation time for $T(n)$ steps is therefore:
        $$
            T(n) \cdot \mathcal{O}(T(n)) = \mathcal{O}(T(n)^2)
        $$
    \end{itemize}
\end{remark}

\begin{example}
    \textbf{Separation: Palindrome Recognition ($L_{pal}$):}
    Consider the palindrome language $L_{pal} = \{ w \in \{0, 1\}^* \mid w = w^R \}$:
    \begin{itemize}
        \item \textbf{2-Tape TM:} Copies $w$ to Tape 2 in $n$ steps, moves head 2 back to the left in $n$ steps, and compares Tape 1 and Tape 2 in $n$ steps. Total time: $\mathcal{O}(n)$ (linear time).
        \item \textbf{1-Tape TM:} Must shuttle back and forth between the opposing ends of $w$ to compare characters, which takes $\Omega(n^2)$ steps. (Formally proven via crossing sequences).
    \end{itemize}
    This proves that a 2-tape TM can achieve a quadratic speedup over a 1-tape TM, and the $\mathcal{O}(T(n)^2)$ simulation bound is asymptotically tight.
\end{example}

\begin{remark}
    \textbf{Invariance of Class $\mathbf{P}$:} Because the overhead of simulating multi-tape TMs on single-tape TMs is polynomial ($\mathcal{O}(T(n)^2)$), the fundamental complexity class $\mathbf{P}$ (problems solvable in polynomial time) is invariant under the number of tapes:
    $$
        \mathbf{P}_{1\text{-tape}} = \mathbf{P}_{k\text{-tape}}
    $$
\end{remark}

\subsection{Nondeterministic Turing Machines}

\begin{definition}
    \textbf{Nondeterministic Turing Machine (NTM):} An NTM is defined similarly to a deterministic TM, but its transition function allows multiple alternative transitions for any given state and tape symbol:
    $$
        \delta : (Q \setminus \{q_{accept}, q_{reject}\}) \times \Gamma \to \mathcal{P}(Q \times \Gamma \times \{\mathrm{L}, \mathrm{R}\})
    $$
    At each step, the machine may choose any transition from $\delta(q, a)$. If $\delta(q, a) = \emptyset$, that computation branch halts without accepting.
\end{definition}

\begin{definition}
    \textbf{Computation Tree and Acceptance:} The set of all possible computations on an input $x$ forms a \textbf{computation tree} rooted at the start configuration $C_0(x)$.
    An NTM $M$ \textbf{accepts} $x$ if \textbf{at least one} computation path in the tree halts in $q_{accept}$:
    $$
        x \in L(M) \iff \exists \text{ path s.t. } C_0(x) \vdash_M^* u q_{accept} v \quad (\text{for some } u, v \in \Gamma^*)
    $$
    $M$ rejects $x$ if all computation branches halt in $q_{reject}$ (or fail to reach $q_{accept}$).
\end{definition}

\begin{theorem}
    \textbf{Equivalence of DTMs and NTMs:} For every nondeterministic Turing machine $N$, there exists an equivalent deterministic Turing machine $D$ such that:
    $$
        L(D) = L(N)
    $$
    Hence, nondeterminism does not increase the computational power of Turing machines (the class of recognizable languages remains identical).
\end{theorem}

\begin{remark}
    \textbf{BFS Simulation and Time Complexity:}
    \begin{itemize}
        \item \textbf{Breadth-First Search (BFS):} The deterministic simulator $D$ systematically explores the computation tree of $N$ level by level. A depth-first search would not suffice, as it could get trapped in an infinite non-accepting branch.
        \item \textbf{Exponential Overhead:} If the shortest accepting path has length $T(n)$ and the maximum branching factor is $b$, the tree contains up to $b^{T(n)}$ configurations. The deterministic simulation runs in \textbf{exponential time} $\mathcal{O}(b^{T(n)})$, which motivates the $\mathbf{P} \overset{?}{=} \mathbf{NP}$ question.
    \end{itemize}
\end{remark}

\subsection{Recognizability and Decidability}

\begin{definition}
    \textbf{Accepted Language:} A word $x \in \Sigma^*$ is accepted by $M$ if the computation of $M$ on $x$ halts in $q_{accept}$:
    $$
        L(M) = \{ x \in \Sigma^* \mid C_0(x) \vdash_M^* u q_{accept} v \text{ for some } u, v \in \Gamma^* \}
    $$
\end{definition}

\begin{definition}
    \textbf{Decidable (Recursive) Language:} A language $L \subseteq \Sigma^*$ is called \textbf{decidable} (or \textbf{recursive}) if there exists a Turing Machine $M$ that \textbf{halts on every input} $x \in \Sigma^*$ such that:
    $$
        x \in L \iff M \text{ halts in } q_{accept}
    $$
    $$
        x \notin L \iff M \text{ halts in } q_{reject}
    $$
    A Turing machine with this property is called a \textbf{decider}.
\end{definition}

\begin{definition}
    \textbf{Turing-Recognizable (Recursively Enumerable) Language:} A language $L \subseteq \Sigma^*$ is called \textbf{Turing-recognizable} (or \textbf{recursively enumerable} / semi-decidable) if there exists a Turing Machine $M$ such that:
    $$
        L(M) = L
    $$
    For words $x \notin L$, $M$ is allowed to halt in $q_{reject}$ or loop indefinitely.
\end{definition}

\begin{remark}
    Every decidable language is Turing-recognizable, but the converse does not hold. For example, the Halting Problem $H = \{ \langle M, w \rangle \mid M \text{ halts on } w \}$ is Turing-recognizable but undecidable.
\end{remark}

\begin{theorem}
    \textbf{Decidability via Complements:} A language $L \subseteq \Sigma^*$ is \textbf{decidable} if and only if both $L$ and its complement $L^c = \Sigma^* \setminus L$ are \textbf{Turing-recognizable}:
    $$
        L \in \mathcal{L}_{\text{rec}} \iff L \in \mathcal{L}_{\text{RE}} \land L^c \in \mathcal{L}_{\text{RE}}
    $$
\end{theorem}

\subsection{Universal Turing Machine and Universal Language}

\begin{definition}
    \textbf{Universal Language ($L_{\text{univ}}$):} Let $\langle M \rangle \in \{0, 1\}^*$ denote a binary encoding of a Turing machine $M$. The \textbf{Universal Language} $L_{\text{univ}}$ (also denoted $L_U$ or $A_{\text{TM}}$) consists of all pairs of machine encodings and input words accepted by that machine:
    $$
        L_{\text{univ}} = \{ \langle M \rangle \# w \mid w \in L(M) \}
    $$
\end{definition}

\subsection{The Diagonal Language ($L_{\text{diag}}$) and Its Complement ($L_{\text{diag}}^c$)}

\begin{definition}
    \textbf{Canonical TM Enumeration:} Let $w_1, w_2, w_3, \dots$ be the canonical lexicographic ordering of all words in $\{0, 1\}^*$. Every word $w_i$ can be uniquely interpreted as an encoding of a Turing machine $M_i = M_{w_i}$ (if $w_i$ is not a syntactically valid encoding, $M_i$ is defined as a dummy TM with $L(M_i) = \emptyset$).
\end{definition}

\begin{definition}
    \textbf{Diagonal Language ($L_{\text{diag}}$):} The diagonal language contains all words $w_i$ whose corresponding Turing machine $M_i$ does \textbf{not} accept its own encoding $w_i$:
    $$
        L_{\text{diag}} = \{ w_i \in \{0, 1\}^* \mid w_i \notin L(M_i) \}
    $$
\end{definition}

\begin{theorem}
    \textbf{Non-Recognizability of $L_{\text{diag}}$ (Diagonalization):} The diagonal language is \textbf{not Turing-recognizable}:
    $$
        L_{\text{diag}} \notin \mathcal{L}_{\text{RE}}
    $$
\end{theorem}

\begin{proof}
    Assume for contradiction that $L_{\text{diag}}$ is Turing-recognizable. Then there exists a Turing machine $M_d$ such that $L(M_d) = L_{\text{diag}}$. Since every TM appears in the enumeration, $M_d = M_{w_d}$ for some word $w_d$. Now consider the input $w_d$:
    $$
        w_d \in L_{\text{diag}} \iff w_d \notin L(M_{w_d}) \iff w_d \notin L_{\text{diag}}
    $$
    This is an immediate contradiction ($X \iff \neg X$). Hence, no such machine $M_d$ can exist.
\end{proof}

\begin{definition}
    \textbf{Complement of $L_{\text{diag}}$ ($L_{\text{diag}}^c$, Self-Acceptance):} The complement of $L_{\text{diag}}$ consists of all words $w_i$ whose corresponding Turing machine $M_i$ \textbf{accepts} $w_i$:
    $$
        L_{\text{diag}}^c = \{0, 1\}^* \setminus L_{\text{diag}} = \{ w_i \in \{0, 1\}^* \mid w_i \in L(M_i) \}
    $$
\end{definition}

\begin{theorem}
    \textbf{Properties of $L_{\text{diag}}^c$:}
    \begin{itemize}
        \item \textbf{Turing-Recognizable ($L_{\text{diag}}^c \in \mathcal{L}_{\text{RE}}$):} On input $w_i$, a machine can decode $w_i$, construct $M_i$, and simulate $M_i$ on input $w_i$ using a universal Turing machine. If $M_i$ accepts $w_i$, the simulation halts and accepts.
        \item \textbf{Undecidable ($L_{\text{diag}}^c \notin \mathcal{L}_{\text{rec}}$):} If $L_{\text{diag}}^c$ were decidable, its complement $(L_{\text{diag}}^c)^c = L_{\text{diag}}$ would also be recognizable by the complement theorem, contradicting $L_{\text{diag}} \notin \mathcal{L}_{\text{RE}}$.
    \end{itemize}
\end{theorem}
